# ---------------------------------------------------------------- subjects

fn join(parts: List[String]) -> String = {
  var out = ""
  for p in parts { out = out ++ p ++ "\n" }
  out
}

fn show_diags(ds: List[Diag]) -> String = {
  var out = ""
  for d in ds { out = out ++ d.msg ++ " | " }
  out
}

## One module per class. Support declarations are a fixed O(1) prefix; the
## per-body cost divides by the number of bodies the scheduler actually ran.
pub fn subject(kind: String, n: Int) -> String = {
  var parts: List[String] = []
  if kind == "scalar" {
    for i in 0..n { parts = parts ++ ["fn s${i}() -> Int = ${i}"] }
  }
  if kind == "arith" {
    for i in 0..n {
      parts = parts ++ ["fn p${i}(a: Int, b: Int) -> Int = {\n  let c = a + b * 2\n  let d = c - a + ${i}\n  let e = if d > 0 { d } else { 0 - d }\n  e + c * 3 - d\n}"]
    }
  }
  if kind == "calls" {
    parts = parts ++ ["fn helper(x: Int) -> Int = x + 1", "fn helper2(x: Int, y: Int) -> Int = x * y - 1"]
    for i in 0..n {
      parts = parts ++ ["fn c${i}(a: Int) -> Int = helper(a) + helper2(a, ${i}) + helper(a + ${i})"]
    }
  }
  if kind == "generic" {
    parts = parts ++ ["trait Scale[T] { fn scale(x: T, k: Int) -> Int }",
      "type Coin = { value: Int }",
      "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Int = x.value * k }"]
    for i in 0..n {
      parts = parts ++ ["fn g${i}[T: Scale](x: T, y: T) -> Int = scale(x, ${i}) + scale(y, 2) + scale(x, 3)"]
    }
  }
  if kind == "inferred" {
    for i in 0..n {
      parts = parts ++ ["fn q${i}(a: Int, b: Int) = {\n  let add = (x: Int, y: Int) => x + y\n  let twice = (f: fn(Int, Int) -> Int, x: Int) => f(x, x)\n  let t = twice(add, a + ${i})\n  let u = twice((x, y) => x * y, b)\n  t + u\n}"]
    }
  }
  if kind == "method" {
    parts = parts ++ ["trait Shine[T] { fn shine(x: T) -> Int }"]
    for i in 0..n {
      parts = parts ++ ["type W${i} = { v: Int }", "impl Shine[W${i}] { fn shine(x: W${i}) -> Int = x.v * 2 + ${i} }"]
    }
  }
  if kind == "default" {
    for i in 0..n {
      parts = parts ++ ["trait Buff${i}[T] { fn raw${i}(x: T) -> Int\n  fn buff${i}(x: T) -> Int = raw${i}(x) + ${i} }"]
    }
  }
  if kind == "java" {
    parts = parts ++ ["use java \"java.lang.Math\""]
    for i in 0..n {
      parts = parts ++ ["fn j${i}(x: Int) -> Int !io = Math.abs(x - ${i}) + Math.max(x, ${i})"]
    }
  }
  join(parts)
}

fn empty_env() -> Map[String, ModExports] = map.empty()

fn base_cx() -> Cx !io = Cx { ..cx_new(Some("sample"), Some("sample.dawn")), jsig: jsig_real() }

fn scope_key() -> ModuleKey = identity.module_key("bench", "sample", "sample", "sample.dawn")

## One parse per revision: the snapshot keeps the tree the headers are checked
## from, beside the code-point index replay projects through.
fn snapshot_of(text: String) -> Snapshot = match source_snapshot.of(text) {
  Some(s) -> s, None -> panic("subject parse")
}

fn headers_of_snapshot(s: Snapshot) -> ModuleHeaders !io = {
  let h = checker.check_module_headers(base_cx(), source_snapshot.syntax(s), empty_env())
  if h.cx.diags != [] { panic("subject headers: ${show_diags(h.cx.diags)}") }
  h
}

fn headers_of(text: String) -> ModuleHeaders !io = headers_of_snapshot(snapshot_of(text))

fn body_count(b: ModuleBodies) -> Int =
  len(b.functions) + len(b.constants) + len(b.impl_methods) + len(b.trait_defaults) + len(b.tests)

# ---------------------------------------------------------------- reporting

fn meta(key: String, value: String) -> Unit !io = println("meta\t${key}\t${value}")

fn row(index: Int, metric: String, nanos: Int) -> Unit !io =
  println("round\t${index}\t${metric}\t${nanos}")

# ---------------------------------------------------------------- phase split

## The counters below mirror `check/scalar_replay`'s own steps one for one.
## `shape` is its per-body admission guard: the header half of class
## membership, the declaration pairing, the relocated signature, the alias
## guard over the recorded binder names and the reserved allocation interval.
## `proj` is its source projection plus `body_product.project`, `reval` its
## `checker.revalidate_read` loop, `asm` its `body_product.assemble`.
##
## Everything the recorded revision can settle alone now lives in `admit`,
## which runs once per recording. The mirror keeps that split: `recorded` and
## `admitted_entries` below are the record-time half and are built outside the
## timed region, as `record` itself is.
##
## This mirror is the fixture's only real obligation, and it is checked: every
## run compares its reused/checked counts against production replay in the same
## process. An admission rule that lands in the executor without landing here
## makes the split panic rather than report a split of something else.
type Acc = { checked: Int, reused: Int, shape: Int, reval: Int, proj: Int, asm: Int }

type Prior = { syntax: FnDecl, signature: Sig, product: Product, declaration: Int,
  binders: List[String], symbols: Map[Int, Sym] }
type Prepared = { scope: ModuleKey, entries: Map[DeclKey, Prior], allowed: Map[DeclKey, Int],
  reserver: allocation.Reserver,
  old_tokens: source_projection.Tokens, tokens: source_projection.Tokens }

fn primitive(t: Ty) -> Bool = match t {
  TyInt -> true, TyFloat -> true, TyBool -> true, TyUnit -> true, _ -> false
}

fn parameters(d: FnDecl, s: Sig) -> Option[Set[String]] = {
  if d.tparams != [] || d.eparams != [] || d.effs != [] || d.ret == None ||
    s.tparams != [] || s.eparams != [] || len(s.constraints) != 0 || s.eff != EPure ||
    s.inferring || s.is_builtin || s.trait_id != None || not primitive(s.ret) ||
    len(d.params) != len(s.param_tys) || len(d.params) != len(s.param_names) { return None }
  for present in s.param_defaults { if present { return None } }
  var bound: Set[String] = set.empty()
  var i = 0
  for param in d.params {
    if param.default != None || param.name != s.param_names[i] ||
      not primitive(s.param_tys[i]) || set.has(bound, param.name) { return None }
    bound = set.insert(bound, param.name)
    i = i + 1
  }
  Some(bound)
}

fn table(scope: ModuleKey, h: ModuleHeaders) -> Option[allocation.Table] =
  allocation.combine([allocation.compiler_headers(scope.world)?,
    allocation.table(allocation.local_headers(scope, h.syntax, h.cx)?)?])

fn snapshot_matches(h: ModuleHeaders, s: Snapshot) -> Bool = source_snapshot.syntax(s) == h.syntax

## The record-time half of admission, mirrored from `scalar_replay.recorded`.
fn recorded(prior: FnDecl, prior_sig: Sig, p: Product, declaration: Int) -> Option[Prior] = {
  if p.tree.sig != prior_sig || p.tree.name != prior.name ||
    p.signatures != [] || p.aliases != [] || len(p.bounds) != 0 ||
    p.diagnostics != [] { return None }
  let reads = p.function_reads?
  if reads == [] { return None }
  for read in reads {
    match read {
      AssignableType(a, b, true) -> if not primitive(a) || not primitive(b) { return None }
      _ -> return None
    }
  }
  var symbols: Map[Int, Sym] = map.empty()
  for change in p.symbols {
    let symbol = change.value?
    if change.key < p.allocation_start || change.key >= p.allocation_start + p.allocation_count ||
      map.has(symbols, change.key) || not primitive(symbol.ty) || symbol.mutable ||
      symbol.dict_of != None || symbol.ev_of != None || symbol.cell_of != None { return None }
    symbols = map.insert(symbols, change.key, symbol)
  }
  var written: Set[Int] = set.empty()
  for write in p.body_writes? {
    match write {
      SymbolKey(id) -> {
        if not map.has(symbols, id) { return None }
        written = set.insert(written, id)
      }
      _ -> return None
    }
  }
  if set.len(written) != map.len(symbols) { return None }
  let names = scalar_shape.binders(prior.body, prior_sig.param_names)?
  Some(Prior { syntax: prior, signature: prior_sig, product: p, declaration: declaration,
    binders: names, symbols: symbols })
}

fn admitted_entries(scope: ModuleKey, old: ModuleHeaders, saved: Recorded) -> Map[DeclKey, Prior] = {
  var originals: Map[DeclKey, (FnDecl, Int)] = map.empty()
  for entry in identity.declarations(scope, old.syntax) {
    match old.syntax.decls[entry.declaration] {
      DFn(d) -> { originals = map.insert(originals, entry.key, (d, entry.declaration)) }
      _ -> ()
    }
  }
  var entries: Map[DeclKey, Prior] = map.empty()
  var seen: Set[DeclKey] = set.empty()
  for entry in saved.entries {
    match entry {
      SavedFunction(k, FunctionInput(prior, prior_sig, false), product) -> {
        if set.has(seen, k) {
          entries = map.remove(entries, k)
        } else {
          seen = set.insert(seen, k)
          match map.get(originals, k) {
            Some((declared, declaration)) -> {
              if declared == prior && map.get(old.cx.fns, prior.name) == Some(prior_sig) {
                match recorded(prior, prior_sig, product, declaration) {
                  Some(value) -> { entries = map.insert(entries, k, value) }
                  None -> ()
                }
              }
            }
            None -> ()
          }
        }
      }
      _ -> ()
    }
  }
  entries
}

## Per-body admission guard, split out so its cost can be read separately.
## Returns the reserved relocation and the recorded local symbols only when
## every structural precondition of replay holds.
type Shaped = { prior: FnDecl, product: Product, symbols: Map[Int, Sym], ids: relocate.Ids,
  prior_declaration: Int, declaration_new: Int }

fn shape_ok(p: Prepared, cx: Cx, d: FnDecl, sig: Sig) -> Option[Shaped] = {
  if cx.diags != [] { return None }
  let bound = parameters(d, sig)?
  let key = DeclKey { scope: p.scope, path: [Named(FunctionDecl, d.name)] }
  let declaration = map.get(p.allowed, key)?
  let saved = map.get(p.entries, key)?
  let prior = saved.syntax
  let prior_sig = saved.signature
  if prior_sig.param_names != sig.param_names { return None }
  if not scalar_shape.same(prior.body, d.body, bound) { return None }
  for name in saved.binders {
    if map.has(cx.module_aliases, name) { return None }
  }
  let ids = allocation.reserved_plan(p.reserver, key, saved.product.allocation_start,
    saved.product.allocation_count, cx.next_id, prior_sig, sig, [])?
  Some(Shaped { prior: prior, product: saved.product, symbols: saved.symbols, ids: ids,
    prior_declaration: saved.declaration, declaration_new: declaration })
}

fn timed_function(p: Prepared, acc: Acc, cx: Cx, d: FnDecl, sig: Sig,
  cold: BodyExecutor[Acc]) -> (Acc, Cx, TFun) !io = {
  let t0 = System.nanoTime()
  let shaped = shape_ok(p, cx, d, sig)
  let t1 = System.nanoTime()
  var a = Acc { ..acc, shape: acc.shape + (t1 - t0) }
  match shaped {
    None -> cold.function(Acc { ..a, checked: a.checked + 1 }, cx, d, sig)
    Some(saved) -> {
      let prior = saved.prior
      let t2 = System.nanoTime()
      let positions = source_projection.pair(p.old_tokens, p.tokens,
        saved.prior_declaration, saved.declaration_new, prior.lo, prior.hi, d.lo, d.hi)
      let moved = match positions {
        None -> None
        Some(pos) -> {
          let view = View { ids: saved.ids,
            positions: (at) => source_projection.start_at(pos, at),
            ends: (at) => source_projection.end_at(pos, at),
            symbols: saved.symbols,
            assertion: (text, lo, hi) => source_projection.assertion_at(pos, text, lo, hi),
            callee: (owner, name, builtin, trait_id) => None }
          body_product.project(view, saved.product, cx.next_id)
        }
      }
      let t3 = System.nanoTime()
      a = Acc { ..a, proj: a.proj + (t3 - t2) }
      match moved {
        None -> cold.function(Acc { ..a, checked: a.checked + 1 }, cx, d, sig)
        Some(product) -> {
          let t4 = System.nanoTime()
          var ok = true
          match product.function_reads {
            None -> { ok = false }
            Some(reads) -> {
              for read in reads {
                if checker.revalidate_read(cx, read) != Some(true) { ok = false }
              }
            }
          }
          let t5 = System.nanoTime()
          a = Acc { ..a, reval: a.reval + (t5 - t4) }
          if not ok {
            cold.function(Acc { ..a, checked: a.checked + 1 }, cx, d, sig)
          } else {
            let carried = BodyProduct { ..product, in_test: cx.in_test, const_cutoff: cx.const_cutoff,
              loop_jumps: cx.loop_jumps, take_cell: cx.take_cell,
              function_reads: match cx.function_reads { None -> None, Some(_) -> product.function_reads },
              body_writes: match cx.body_writes { None -> None, Some(_) -> product.body_writes },
              frame: Frame { ..product.frame, isolated: cx.frame.isolated } }
            let t6 = System.nanoTime()
            let assembled = body_product.assemble(cx, carried)
            let t7 = System.nanoTime()
            a = Acc { ..a, asm: a.asm + (t7 - t6) }
            match assembled {
              Some(after) -> (Acc { ..a, reused: a.reused + 1 }, after, carried.tree)
              None -> cold.function(Acc { ..a, checked: a.checked + 1 }, cx, d, sig)
            }
          }
        }
      }
    }
  }
}

type Split = { bind: Int, pair: Int, alloc: Int, acc: Acc, total: Int, bodies: Int }

fn instrumented_replay(scope: ModuleKey, old_headers: allocation.Table,
  entries: Map[DeclKey, Prior], old_source: Snapshot, next: ModuleHeaders, source: Snapshot) -> Split !io = {
  let start = System.nanoTime()
  let p0 = System.nanoTime()
  let ok = next.cx.diags == [] && snapshot_matches(next, source)
  let p1 = System.nanoTime()
  if not ok { panic("instrumented replay: source snapshots did not match") }
  let a0 = System.nanoTime()
  let next_table = table(scope, next)
  let reserver = match next_table {
    Some(b) -> allocation.reserver(old_headers, b)
    None -> None
  }
  let a1 = System.nanoTime()
  let prepared_reserver = match reserver { Some(v) -> v, None -> panic("instrumented replay: no allocation plan") }
  let q0 = System.nanoTime()
  var allowed: Map[DeclKey, Int] = map.empty()
  for entry in identity.declarations(scope, next.syntax) {
    allowed = map.insert(allowed, entry.key, entry.declaration)
  }
  let q1 = System.nanoTime()
  let prepared = Prepared { scope: scope, entries: entries, allowed: allowed,
    reserver: prepared_reserver,
    old_tokens: source_snapshot.tokens(old_source), tokens: source_snapshot.tokens(source) }
  let cold: BodyExecutor[Acc] = checker.cold_body_executor()
  let executor: BodyExecutor[Acc] = BodyExecutor {
    function: (acc, cx, d, sig) => timed_function(prepared, acc, cx, d, sig, cold),
    inferred_body: (a, cx, d, sig) => cold.inferred_body(Acc { ..a, checked: a.checked + 1 }, cx, d, sig),
    constant: (a, cx, d, ty, visible) => cold.constant(Acc { ..a, checked: a.checked + 1 }, cx, d, ty, visible),
    method: (a, cx, tr, subject, d, sig) => cold.method(Acc { ..a, checked: a.checked + 1 }, cx, tr, subject, d, sig),
    default_body: (a, cx, tr, d, sig, body) => cold.default_body(Acc { ..a, checked: a.checked + 1 }, cx, tr, d, sig, body),
    test_body: (a, cx, name, body) => cold.test_body(Acc { ..a, checked: a.checked + 1 }, cx, name, body)
  }
  let zero = Acc { checked: 0, reused: 0, shape: 0, reval: 0, proj: 0, asm: 0 }
  let (acc, bodies) = checker.execute_module_bodies(next, zero, executor)
  let stop = System.nanoTime()
  Split { bind: p1 - p0, pair: q1 - q0, alloc: a1 - a0, acc: acc,
    total: stop - start, bodies: body_count(bodies) }
}

# ---------------------------------------------------------------- modes

fn edited(text: String) -> String = "# edited elsewhere in the file\n" ++ text

fn run_cold(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  let headers = headers_of(text)
  var executed = 0
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let bodies = checker.check_module_body_products(headers)
    let t1 = System.nanoTime()
    if bodies.cx.diags != [] { panic("cold bodies: ${show_diags(bodies.cx.diags)}") }
    executed = body_count(bodies)
    row(r, "cold", t1 - t0)
  }
  meta("bodies", "${executed}")
  meta("source_bytes", "${str.len(text)}")
}

fn run_record(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  let snapshot = snapshot_of(text)
  let headers = headers_of_snapshot(snapshot)
  let scope = scope_key()
  var executed = 0
  var refused = 0
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let saved = body_execution.record(scope, headers)
    let t1 = System.nanoTime()
    let admitted = scalar_replay.admit(scope, headers, snapshot, saved)
    let t2 = System.nanoTime()
    if saved.bodies.cx.diags != [] { panic("record bodies: ${show_diags(saved.bodies.cx.diags)}") }
    if scalar_replay.admitted_count(admitted) < 0 { panic("admission verdict") }
    executed = body_count(saved.bodies)
    refused = saved.refused
    row(r, "record", t1 - t0)
    row(r, "admit", t2 - t1)
  }
  meta("bodies", "${executed}")
  meta("refused", "${refused}")
}

type Setup = { scope: ModuleKey, old: ModuleHeaders, before: Snapshot, saved: Recorded,
  admitted: scalar_replay.Admitted, entries: Map[DeclKey, Prior], old_headers: allocation.Table,
  next: ModuleHeaders, after: Snapshot }

## One revision pair: the same module, edited only by a leading comment. Every
## body keeps its text and its declaration identity, and every byte offset moves.
## Both snapshots are built here, outside the timed region: each is the single
## parse its own revision's headers were checked from.
fn replay_setup(kind: String, n: Int) -> Setup !io = {
  let before = snapshot_of(subject(kind, n))
  let after = snapshot_of(edited(subject(kind, n)))
  let scope = scope_key()
  let old = headers_of_snapshot(before)
  let saved = body_execution.record(scope, old)
  if saved.refused != 0 { panic("replay setup: ${saved.refused} refused recordings") }
  Setup { scope: scope, old: old, before: before, saved: saved,
    admitted: scalar_replay.admit(scope, old, before, saved),
    entries: admitted_entries(scope, old, saved),
    old_headers: match table(scope, old) { Some(t) -> t, None -> panic("replay setup: no old table") },
    next: headers_of_snapshot(after), after: after }
}

fn run_replay(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let s = replay_setup(kind, n)
  var reused = 0
  var checked = 0
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let result = scalar_replay.replay(s.admitted, s.next, s.after)
    let t1 = System.nanoTime()
    reused = result.reused
    checked = result.checked
    row(r, "replay", t1 - t0)
  }
  meta("reused", "${reused}")
  meta("checked", "${checked}")
}

## The phase split runs in its own JVM so it gets the same warmup treatment as
## the production entry it decomposes; the two totals are then comparable.
fn run_split(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let s = replay_setup(kind, n)
  let control = scalar_replay.replay(s.admitted, s.next, s.after)
  var bodies = 0
  for r in 0..rounds {
    let split = instrumented_replay(s.scope, s.old_headers, s.entries, s.before, s.next, s.after)
    if split.acc.reused != control.reused || split.acc.checked != control.checked {
      panic("instrumented split disagrees: ${split.acc.reused}/${split.acc.checked} vs ${control.reused}/${control.checked}")
    }
    bodies = split.bodies
    row(r, "split_total", split.total)
    row(r, "split_bind", split.bind)
    row(r, "split_pair", split.pair)
    row(r, "split_alloc", split.alloc)
    row(r, "split_shape", split.acc.shape)
    row(r, "split_reval", split.acc.reval)
    row(r, "split_proj", split.acc.proj)
    row(r, "split_asm", split.acc.asm)
  }
  meta("reused", "${control.reused}")
  meta("checked", "${control.checked}")
  meta("bodies", "${bodies}")
}

## The two halves of snapshot provenance, priced apart.
##
## `snapshot_build` is what the revision's owner pays once, and it parses; it
## is not replay's cost, because the owner had to parse the revision anyway to
## check its headers. `bind_pair` is what replay still pays before it looks at
## a single body: two tree comparisons against the headers each snapshot
## accompanies. `parse` and `index` are `snapshot_build`'s two parts, and
## `parse` by itself is what a reparsing replay used to pay twice per call.
fn run_parse(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  let headers = headers_of(text)
  for r in 0..rounds {
    let t0 = System.nanoTime()
    let (syntax, ds) = parser.parse_module(text)
    let t1 = System.nanoTime()
    let indexed = source_projection.index(text)
    let t2 = System.nanoTime()
    let same = ds == [] && syntax == headers.syntax
    let t3 = System.nanoTime()
    if not same { panic("reparse did not match the header syntax") }
    if match source_projection.between_indexed(indexed, indexed, 0, 0, 0, 0) {
      Some(_) -> false, None -> true
    } { panic("index did not bind to its own snapshot") }
    let t4 = System.nanoTime()
    let whole = snapshot_of(text)
    let t5 = System.nanoTime()
    if source_snapshot.syntax(whole) != headers.syntax { panic("snapshot tree did not match the headers") }
    row(r, "parse", t1 - t0)
    row(r, "index", t2 - t1)
    row(r, "compare", t3 - t2)
    row(r, "snapshot_build", t2 - t0)
    row(r, "snapshot_of", t5 - t4)
    row(r, "bind_pair", 2 * (t3 - t2))
  }
  meta("source_bytes", "${str.len(text)}")
  meta("decls", "${len(headers.syntax.decls)}")
}

## Retained cost, attributed rather than estimated: settle the heap, read used
## bytes, retain one more value, settle and read again.
##
## Two metrics, because only one of them is new memory. `retained_index` is the
## code-point index, which nothing retained before: replay built it per call
## and dropped it. `retained_snapshot` also builds a fresh tree, which the real
## owner does not: there the snapshot holds the same tree the headers hold, so
## the index is the whole marginal cost and the difference between the two
## numbers is the tree this fixture parsed a second time.
fn used_heap() -> Int !io = {
  let rt = Runtime.getRuntime().expect("runtime")
  var i = 0
  while i < 4 {
    System.gc()
    i = i + 1
  }
  rt.totalMemory() - rt.freeMemory()
}

fn run_memory(kind: String, n: Int, rounds: Int) -> Unit !io = {
  let text = subject(kind, n)
  var indexes: List[source_projection.Indexed] = []
  for r in 0..rounds {
    let before = used_heap()
    indexes = indexes ++ [source_projection.index(text)]
    let after = used_heap()
    row(r, "retained_index", after - before)
  }
  var keep: List[Snapshot] = []
  for r in 0..rounds {
    let before = used_heap()
    keep = keep ++ [snapshot_of(text)]
    let after = used_heap()
    row(r, "retained_snapshot", after - before)
  }
  if len(keep) != rounds || len(indexes) != rounds { panic("values were not retained") }
  meta("source_bytes", "${str.len(text)}")
}

## Cost of the nanoTime pairs the split adds, so the split can be discounted.
fn run_timer(kind: String, n: Int, rounds: Int) -> Unit !io = {
  for r in 0..rounds {
    let t0 = System.nanoTime()
    var sink = 0
    for i in 0..n {
      let a = System.nanoTime()
      let b = System.nanoTime()
      sink = sink + (b - a)
    }
    let t1 = System.nanoTime()
    if sink < 0 { panic("timer sink") }
    row(r, "timer", t1 - t0)
  }
  meta("pairs", "${n}")
}

pub fn main() -> Unit !io = {
  let a = args()
  if len(a) < 3 { panic("usage: <cold|record|replay|split|parse|memory|timer> <class> <n> [rounds]") }
  let mode = a[0]
  let kind = a[1]
  let n = Integer.parseInt(a[2])
  let rounds = if len(a) > 3 { Integer.parseInt(a[3]) } else { 11 }
  meta("mode", mode)
  meta("class", kind)
  meta("n", "${n}")
  meta("rounds", "${rounds}")
  meta("java", match System.getProperty("java.version") { Some(v) -> v, None -> "unknown" })
  if mode == "cold" { run_cold(kind, n, rounds) }
  else { if mode == "record" { run_record(kind, n, rounds) }
  else { if mode == "replay" { run_replay(kind, n, rounds) }
  else { if mode == "split" { run_split(kind, n, rounds) }
  else { if mode == "parse" { run_parse(kind, n, rounds) }
  else { if mode == "memory" { run_memory(kind, n, rounds) }
  else { if mode == "timer" { run_timer(kind, n, rounds) }
  else { panic("unknown mode ${mode}") } } } } } } }
}
