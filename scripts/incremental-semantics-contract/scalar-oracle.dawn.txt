# Private fixture: the Java counter is injected into canonical checker entries,
# never into the executor's self-reported counters or its admission decisions.
use compiler/check/checker
use compiler/check/checker.{ModuleHeaders, ModuleBodies}
use compiler/check/body_execution
use compiler/check/scalar_replay
use compiler/check/source_snapshot
use compiler/check/cx.{Cx, cx_new}
use compiler/check/identity.{module_key}
use compiler/check/semantic_reads
use std/map
use java "contract.ScalarReplaySnapshot"

pub type Comparison = { old: ModuleBodies, current: ModuleBodies }
pub fn sample_count(xs: List[Comparison]) -> Int = len(xs)
pub fn sample_at(xs: List[Comparison], index: Int) -> Comparison = xs[index]
pub fn main() -> Unit !io = { let pairs = samples()
  if len(pairs) != 32 { panic("scalar pair count") }
}

pub fn samples() -> List[Comparison] !io = {
  let support = "const A: Int = 1\nfn leaf() = A\n" ++
    "type Coin = { value: Int }\n" ++
    "trait Waxy[T] { fn wax(x: T) -> Int\n fn buff(x: T) -> Int = wax(x) }\n" ++
    "impl Waxy[Coin] { fn wax(x: Coin) -> Int = x.value }\n" ++
    "fn other(x: Int) -> Int = x\ntest \"sample\" { assert leaf() == 1 }\n"
  let before = "fn answer() -> Int = 42\n" ++ support
  let cases = [
    ("# moved\nfn answer() -> Int = 42\n", 2),
    ("type Added = { n: Int }\nfn answer() -> Int = 42\n", 2),
    ("fn answer() -> Int = 43\n", 1),
    ("fn answer() -> Bool = true\n", 1),
    ("fn answer() -> Int = { 42 }\n", 1),
    ("fn answer() = 42\n", 1),
    ("fn answer() -> Int = true\n", 1),
    ("fn answer() -> Int = 42\nfn answer() -> Int = 42\n", 0)
  ]
  let scope = module_key("scalar-oracle", "sample", "sample", "sample.dawn")
  let base = cx_new(Some("sample"), Some("sample.dawn"))
  let old_snapshot = match source_snapshot.of(before) {
    Some(s) -> s, None -> panic("old scalar fixture parse")
  }
  let old = checker.check_module_headers(base, source_snapshot.syntax(old_snapshot), map.empty())
  let saved = body_execution.record(scope, old)
  if saved.refused != 0 { panic("old scalar capture") }
  var pairs: List[Comparison] = []
  for (tail, hits) in cases {
    let after = support ++ tail
    let next_snapshot = match source_snapshot.of(after) {
      Some(s) -> s, None -> panic("new scalar fixture parse")
    }
    let headers = checker.check_module_headers(base, source_snapshot.syntax(next_snapshot), map.empty())
    for logging in [false, true] {
      let next = ModuleHeaders { ..headers, cx: Cx { ..headers.cx,
        function_reads: if logging { Some([semantic_reads.StdModuleMode(false)]) } else { None },
        body_writes: if logging { Some([]) } else { None } } }
      ScalarReplaySnapshot.reset()
      let current = scalar_replay.replay(scope, old, old_snapshot, saved, next, next_snapshot)
      let calls = ScalarReplaySnapshot.count()
      if current.reused != hits || calls != current.checked {
        panic("FAIL: scalar independent execution count")
      }
      ScalarReplaySnapshot.reset()
      let cold = checker.check_module_body_products(next)
      if ScalarReplaySnapshot.count() != calls + hits { panic("FAIL: scalar cold execution count") }
      pairs = pairs ++ [Comparison { old: cold, current: current.bodies }]
    }
  }
  pairs ++ local_samples() ++ boundary_samples()
}

fn local_samples() -> List[Comparison] !io = {
  let body = "fn compute(x: Int) -> Int = {\n let y = x + 1\n if y > 0 { y * 2 } else { -x }\n}\n"
  let prefix = "type Added = { n: Int }\nfn prefix(a: Int, b: Int) -> Int = a + b\n"
  replay_cases("local", "fn prefix() -> Int = 0\n" ++ body, [
    (prefix ++ "# moved\n" ++ body, 1),
    (body ++ prefix, 1),
    (prefix ++ "fn compute(x: Int) -> Int = {\n let y = x + 2\n if y > 0 { y * 2 } else { -x }\n}\n", 0),
    (prefix ++ "fn compute(x: Int) -> Int = {\n let y = x + 1\n if y > 0 { y * 2 } else { x }\n}\n", 0),
    (prefix ++ "fn compute(x: Int) -> Int = {\n let y = x + 1\n let x = y * 2\n x\n}\n", 0),
    (prefix ++ "fn compute(x: Int) -> Int = {\n let y: Int = x + 1\n if y > 0 { y * 2 } else { -x }\n}\n", 0)
  ])
}

## Newlines carry statement boundaries, and the token projection drops them.
## These two revisions have one token stream and two parses, so only the paired
## syntax trees can tell them apart.
fn boundary_samples() -> List[Comparison] !io = {
  let before = "fn compute(x: Int) -> Int = {\n let a = 1\n let b = a\n - x\n}\n"
  replay_cases("boundary", before, [
    ("# moved\n" ++ before, 1),
    ("fn compute(x: Int) -> Int = {\n let a = 1\n let b = a - x\n}\n", 0)
  ])
}

fn replay_cases(label: String, before: String, cases: List[(String, Int)]) -> List[Comparison] !io = {
  let scope = module_key("scalar-oracle", "sample", "sample", "sample.dawn")
  let base = cx_new(Some("sample"), Some("sample.dawn"))
  let old_snapshot = match source_snapshot.of(before) {
    Some(s) -> s, None -> panic(label ++ " fixture parse")
  }
  let old = checker.check_module_headers(base, source_snapshot.syntax(old_snapshot), map.empty())
  let saved = body_execution.record(scope, old)
  if saved.refused != 0 || saved.bodies.cx.diags != [] { panic(label ++ " capture") }
  var pairs: List[Comparison] = []
  for (after, hits) in cases {
    let next_snapshot = match source_snapshot.of(after) {
      Some(s) -> s, None -> panic(label ++ " candidate parse")
    }
    let headers = checker.check_module_headers(base, source_snapshot.syntax(next_snapshot), map.empty())
    for logging in [false, true] {
      let next = ModuleHeaders { ..headers, cx: Cx { ..headers.cx,
        function_reads: if logging { Some([semantic_reads.StdModuleMode(false)]) } else { None },
        body_writes: if logging { Some([]) } else { None } } }
      ScalarReplaySnapshot.reset()
      let current = scalar_replay.replay(scope, old, old_snapshot, saved, next, next_snapshot)
      let calls = ScalarReplaySnapshot.count()
      if current.reused != hits || calls != current.checked {
        panic("FAIL: scalar independent execution count")
      }
      ScalarReplaySnapshot.reset()
      let cold = checker.check_module_body_products(next)
      if ScalarReplaySnapshot.count() != calls + hits { panic("FAIL: scalar cold execution count") }
      pairs = pairs ++ [Comparison { old: cold, current: current.bodies }]
    }
  }
  pairs
}
