# Which assertions the production mutant reddens, and which one owns it.
#
# `assert` names the roster in the order run.sh checks it; the red set is
# observed by run.sh and compared with this file. The `owner` line is a hand
# edit, because which assertion owns a mutant is a design decision and a
# recorder that could reassign owners would launder the collision it exists to
# catch (scripts/pipe-contract/matrix.txt is the precedent this copies).
#
# `finished` and `record_finished` are controls: ownership scheduling decides
# where dup and drop go, never what either program computes. `hamt_in_place`
# is also the negative control for the record-spread mutant -- a direct map
# accumulator does not have a spread owner to release.

assert	finished
assert	hamt_in_place
assert	record_finished
assert	record_spread_in_place

control	finished
control	record_finished

role	borrow-hamt-assoc-again	counted
owner	borrow-hamt-assoc-again	hamt_in_place
red	borrow-hamt-assoc-again	hamt_in_place
red	borrow-hamt-assoc-again	record_spread_in_place

role	keep-record-spread-source	counted
owner	keep-record-spread-source	record_spread_in_place
red	keep-record-spread-source	record_spread_in_place
