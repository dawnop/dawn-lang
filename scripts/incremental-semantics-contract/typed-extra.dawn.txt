# Spliced into the private fixture expression only for the full typed mapper.
# The assertion in an ordinary function deliberately exercises error recovery;
# Dawn permits assert only in test blocks, so its one diagnostic is expected.
"pub fn matched(x: Option[Int]) -> Int = match x { Some(y) -> y\n None -> 0 }\n" ++
"pub fn ranged(x: Int) -> Int = { var sum = 0\n for n in 0..x { sum = sum + n }\n sum }\n" ++
"pub fn asserted(x: Int) -> Int = { assert x > 0\n x }\n" ++
"pub fn packed[!e](f: fn() -> Int !e) -> Int !Ask !e = polymorphic(() => f() + ask())\n" ++
"pub fn listed(x: Int) -> List[Int] = [x, x + 1]\n" ++
"pub fn tupled(x: Int) -> (Int, Bool) = (x, x > 0)\n" ++
"pub fn chars() -> Char = 'a'\n" ++
"pub fn floats() -> Float = 1.5\n" ++
"pub fn strings(x: Int) -> String = \"value=\$x\"\n" ++
"pub fn unwrapped(x: Option[Int]) -> Int = x!\n" ++
"pub fn propagated(x: Option[Int]) -> Option[Int] = Some(x?)\n" ++
"pub fn returned(x: Int) -> Int = { return x }\n"
