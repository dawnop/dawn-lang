# Collect frozen and factored products through actual scheduler entries. The
# host reads these returned records; Dawn records are not coerced to Java Object.
pub(pkg) type EntryPair = { expected: Cx, actual: Cx, expected_tree: TFun, tree: TFun,
  detached: Cx, repeated: Cx }
type EntryState = { pairs: List[EntryPair] }
pub(pkg) type EntrySuite = { pairs: List[EntryPair], hits: List[Pair] }

fn entry_step(state: EntryState, cx: Cx, d: FnDecl, sig: Sig) -> (EntryState, Cx, TFun) !io = {
  let original_slots = compiler_cx.slot_of(cx)
  let original_symbols = map.len(cx.syms)
  let detached = checker.entry_after_defaults(cx, d, sig)
  let repeated = checker.entry_after_defaults(cx, d, sig)
  if compiler_cx.slot_of(cx) != original_slots || map.len(cx.syms) != original_symbols ||
    detached.parameters != repeated.parameters {
    panic("function entry temporary context advanced the live scheduler")
  }
  let (expected, expected_tree) = checker.entry_frozen_fn(cx, d, sig)
  let (actual, tree) = checker.check_fn(cx, d, sig)
  let dictionaries = checker.function_dictionaries(detached.cx, sig)
  if detached.parameters != tree.param_syms || dictionaries != tree.dict_syms {
    panic("function entry ordered symbols differ from the captured body")
  }
  for id in detached.parameters ++ dictionaries {
    if map.get(detached.cx.syms, id) != map.get(actual.syms, id) {
      panic("function entry symbol metadata differs from the captured body")
    }
  }
  (EntryState { pairs: state.pairs ++ [EntryPair { expected: expected, actual: actual,
    expected_tree: expected_tree, tree: tree, detached: detached.cx, repeated: repeated.cx }] }, actual, tree)
}

pub(pkg) fn entry_cases() -> EntrySuite !io = {
  let scope = identity.module_key("generic-trace", "sample", "sample", "sample.dawn")
  var sources: List[String] = []
  for sample in samples() { sources = sources ++ [sample.source] }
  sources = sources ++ [
    "fn stable(x: Int) -> Int = x\n",
    "effect Ask { fn ask() -> Int }\nfn answer() -> Int !Ask = ask()\n",
    "fn io_body(x: Int) -> Int !io = x\n",
    "fn defaulted(x: Int = 1) -> Int = x\n",
    "fn duplicate(x: Int, x: Int) -> Int = x\n",
    "fn bad_default(x: Int = true) -> Int = x\n",
    "trait Scale[T] { fn scale(x: T, k: Int) -> Int }\nfn bounded[T: Scale](x: T, k: Int = 2) -> Int = scale(x, k)\n",
    "fn invalid(x: Int) -> Int = true\n"
  ]
  var state = EntryState { pairs: [] }
  var hits: List[Pair] = []
  for source in sources {
    let snapshot = source_snapshot.of(source).expect("entry fixture snapshot")
    let base = cx_new(Some("sample"), Some("sample.dawn"))
    let initial = checker.check_module_headers(base, source_snapshot.syntax(snapshot), map.empty())
    if initial.cx.diags != [] { panic("function entry fixture header diagnostics") }
    for observing in [false, true] {
      let headers = ModuleHeaders { ..initial, cx: Cx { ..initial.cx,
        function_reads: if observing { Some([]) } else { None },
        body_writes: if observing { Some([]) } else { None } } }
      let cold: BodyExecutor[EntryState] = checker.cold_body_executor()
      let executor = BodyExecutor { ..cold,
        function: (value, cx, d, sig) => entry_step(value, cx, d, sig),
        method: (value, cx, _trait, _subject, d, sig) => entry_step(value, cx, d, sig) }
      let (next, products) = checker.execute_module_bodies(headers, state, executor)
      state = next
      let malformed = source == "fn duplicate(x: Int, x: Int) -> Int = x\n" ||
        source == "fn bad_default(x: Int = true) -> Int = x\n" ||
        source == "fn invalid(x: Int) -> Int = true\n"
      if malformed {
        if products.cx.diags == [] { panic("function entry malformed case lost its diagnostics") }
      } else if products.cx.diags != [] {
        var details = ""
        for diagnostic in products.cx.diags { details = details ++ diagnostic.msg ++ " | " }
        panic("function entry clean case acquired diagnostics: " ++ source ++ details)
      }
      if source == "fn stable(x: Int) -> Int = x\n" {
        let saved = body_execution.record(scope, headers)
        let admitted = scalar_replay.admit(scope, headers, snapshot, saved)
        let actual = scalar_replay.replay(admitted, headers, snapshot)
        let expected = checker.check_module_body_products(headers)
        if actual.reused != 1 || actual.checked != 0 { panic("function entry detached proof lost actual scalar replay") }
        hits = hits ++ [Pair { cold: expected, actual: actual.bodies }]
      }
    }
  }
  EntrySuite { pairs: state.pairs, hits: hits }
}

pub(pkg) fn entry_pair_count(suite: EntrySuite) -> Int = len(suite.pairs)
pub(pkg) fn entry_pair_at(suite: EntrySuite, index: Int) -> EntryPair = suite.pairs[index]
pub(pkg) fn entry_hit_count(suite: EntrySuite) -> Int = len(suite.hits)
pub(pkg) fn entry_hit_at(suite: EntrySuite, index: Int) -> Pair = suite.hits[index]
