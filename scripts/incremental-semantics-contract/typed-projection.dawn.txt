## Every callee query these fixtures make is answered twice: once through the
## per-revision index production carries, once through the linear scan the
## index replaced. Disagreement is fatal, so the whole typed-projection corpus
## is an equivalence oracle for the index on real header shapes.
pub(pkg) fn checked_callee(cx: Cx, owner: Option[String], name: String, builtin: Bool,
  tr: Option[Int]) -> Option[Sig] = {
  let indexed = callee_index.signature(callee_index.of(cx), cx, owner, name, builtin, tr)
  let scanned = callee_index.scanned_signature(cx, owner, name, builtin, tr)
  if indexed != scanned { panic("callee index disagrees with the linear scan: " ++ name) }
  indexed
}

## The scheduler enters the declaration a body belongs to before checking it
## and leaves it after, so the body's diagnostics are recorded as offsets from
## that declaration rather than from the file. The probe drives `check_fn`
## itself, so it does the same thing here: a product whose diagnostics were
## raised with no owner is one the scheduler could not have produced, and the
## projection refuses it rather than carrying positions it has no anchor for.
## The resolver that turns what a body recorded inside a declaration back
## into positions in the file, which is the other half of opening one.
fn decl_resolver(cx: Cx, m: Module, d: FnDecl) -> tast_positions.Resolver =
  path_resolver(cx, m, [Named(FunctionDecl, d.name)])

fn path_resolver(cx: Cx, m: Module, path: List[Part]) -> tast_positions.Resolver =
  match span_at(declaration_spans(m), path_text(path)) {
    Some(sp) -> tast_positions.owned(sp.lo, sp.hi, cx.src_path, cx.line_starts)
    None -> tast_positions.unowned(cx.src_path, cx.line_starts)
  }

## Leave the declaration again: the symbols it minted, resolved beside the
## tree that reads them, and no declaration open afterwards.
fn left_decl(after: Cx, r: tast_positions.Resolver, from: Int, to: Int) -> Cx =
  unowned(Cx { ..after, syms: tast_positions.symbols(r, after.syms, from, to) })

## A diagnostic raised in a context with no module path to name stays
## absolute (`cx.raised`), so a product replayed onto a declaration that moved
## carries it to where that declaration now is. Everything before `keep` was
## raised in this revision already.
fn shifted_diags(cx: Cx, keep: Int, delta: Int) -> Cx = {
  var out: List[Diag] = []
  var i = 0
  for d in cx.diags {
    out = out ++ [if i < keep { d } else { Diag { ..d, lo: d.lo + delta, hi: d.hi + delta } }]
    i = i + 1
  }
  Cx { ..cx, diags: out }
}

## The declaration a body is checked inside, opened the way the scheduler
## opens it: measured from the declaration's start, and numbering what the
## body binds inside that declaration rather than in the module's free pool.
## A fixture that captures a product has to hold this context on both sides
## of the check, because a capture compares the two.
pub(pkg) fn owned_decl(cx: Cx, m: Module, d: FnDecl) -> Cx =
  owned_path(cx, m, [Named(FunctionDecl, d.name)])

## The same for any declaration the span view can name, which is how a test
## block is opened.
pub(pkg) fn owned_path(cx: Cx, m: Module, path: List[Part]) -> Cx = {
  let decl = path_text(path)
  match span_at(declaration_spans(m), decl) {
    Some(sp) -> {
      # A diagnostic names the view it resolves against by module path, which
      # only a driver supplies; a context without one measures from the
      # declaration and names nothing, which is what these fixtures have.
      let measured = match cx.mod_path {
        None -> measured_from(cx, sp.lo)
        Some(module) -> owned_by(cx, module, decl, sp.lo)
      }
      enter_decl_owner(measured, path, sp.lo, sp.hi)
    }
    None -> cx
  }
}

pub(pkg) fn probe_decl(cx: Cx, m: Module, d: FnDecl) -> Cx = {
  # The owner is named here rather than taken from `cx.mod_path`, which these
  # fixtures leave empty: a diagnostic raised with no owner to name stays
  # absolute (`cx.raised`), and this probe's trials replay a body onto a
  # declaration that moved. The module samples below take the other branch,
  # because they compare against `check_module` on the same empty path.
  let decl = path_text([Named(FunctionDecl, d.name)])
  match span_at(declaration_spans(m), decl) {
    Some(sp) -> enter_decl_owner(owned_by(cx, "probe", decl, sp.lo),
      [Named(FunctionDecl, d.name)], sp.lo, sp.hi)
    None -> cx
  }
}

pub(pkg) fn checked_body(before: Cx, m: Module, d: FnDecl, sig: Sig) -> (Cx, TFun) !io = {
  let (next, tree) = check_fn(probe_decl(before, m, d), d, sig)
  # Still inside the declaration. Production captures a body product before
  # the scheduler leaves, and a capture taken outside reads the free pool's
  # slot count instead of the declaration's and refuses.
  (next, tree)
}

## The tree `checked_body` hands back holds spans measured from the
## declaration, exactly as the product recorded from it does. The scheduler
## turns them into positions at the declaration's exit; a fixture comparing
## one with a cold module has to do the same thing, and this is that step.
pub(pkg) fn resolved_body(cx: Cx, m: Module, d: FnDecl, f: TFun) -> TFun = {
  let decl = path_text([Named(FunctionDecl, d.name)])
  match span_at(declaration_spans(m), decl) {
    Some(sp) -> tast_positions.function(
      tast_positions.owned(sp.lo, sp.hi, cx.src_path, cx.line_starts), f)
    None -> f
  }
}

## A body the fixture checked without entering a declaration measures from the
## file, so replaying it onto a declaration that moved -- and whose text is
## unchanged, which is the only case admission allows -- is one shift of
## everything in it. `width` bounds the offsets the shift will accept, and the
## file's own length is that bound. No source site is derived: this tree left
## the body checker without one, and so did the cold check it is compared
## against.
fn shift(delta: Int, width: Int) -> tast_positions.Resolver =
  tast_positions.owned(delta, delta + width, None, [])

pub(pkg) fn shifted(delta: Int, width: Int, f: TFun) -> TFun = tast_positions.function(shift(delta, width), f)

fn shifted_constant(delta: Int, width: Int, c: TConst) -> TConst =
  tast_positions.constant(shift(delta, width), c)

## What the scheduler would have left behind for a replayed declaration: the
## symbols it minted and the diagnostics it raised, placed where the
## declaration now sits. A fixture that checks bodies without entering a
## declaration measures both from the file, so both move by the same distance.
fn shifted_syms(before: Cx, after: Cx, delta: Int, width: Int) -> Cx =
  shifted_syms_minted(before, after, mint_cursor(before), mint_cursor(after), delta, width)

## The same, for a caller whose two contexts are not the pair that minted the
## symbols. A body's keys are slots of the declaration it was checked inside,
## so the interval that names them comes from that declaration's own cursors
## and not from a context that has since left it.
fn shifted_syms_minted(before: Cx, after: Cx, from: Int, to: Int, delta: Int, width: Int) -> Cx = {
  let moved = tast_positions.symbols(shift(delta, width), after.syms, from, to)
  var diags: List[Diag] = []
  var i = 0
  for d in after.diags {
    diags = diags ++ [if i < len(before.diags) { d } else { Diag { ..d, lo: d.lo + delta, hi: d.hi + delta } }]
    i = i + 1
  }
  Cx { ..after, syms: moved, diags: diags }
}

fn local_view(before: Cx, after: Cx, delta: Int) -> View = {
  View { symbols: after.syms,
    callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
}

fn admitted_body(v: View, f: TFun) -> TFun = {
  if not body_admit.function(v, f) { panic("test body admission failed: " ++ f.name) }
  f
}

## The tree a body recorded, once the candidate revision has said it can
## install it. Nothing in it moves any more, so what used to be a rebuild is
## the admission walk and the recorded tree itself.
pub(pkg) fn body(f: TFun, before: Cx, after: Cx, delta: Int) -> TFun = {
  if not body_admit.function(local_view(before, after, delta), f) {
    panic("typed admission failed: " ++ f.name)
  }
  f
}

## The same projection onto a declaration that moved. Nothing positional
## happens here: the spans are measured from the declaration, so the move is
## in where the tree is resolved, and that is the fixture's own last step.
pub(pkg) fn body_in_source(f: TFun, before: Cx, after: Cx, delta: Int, cx: Cx, m: Module, d: FnDecl) -> TFun = {
  if not body_admit.function(local_view(before, after, delta), f) {
    panic("source-aware typed admission failed: " ++ f.name)
  }
  resolved_body(cx, m, d, f)
}

# The fixture still supplies allocation/provenance tables, but the complete
# write product and its source/ID projection now belong to production code.
pub(pkg) fn source_state(current: Cx, before: Cx, after: Cx, body: TFun, delta: Int) -> Cx = {
  let view = View { symbols: after.syms,
    callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
  let saved = body_product.capture(before, after, body).expect("state capture: " ++ body.name)
  let projected = body_product.project(view, saved).expect("state projection: " ++ body.name)
  body_product.assemble(current, projected).expect("state assembly: " ++ body.name)
}

pub(pkg) type HeaderTrial = { relocated: TFun, cold: TFun, replayed_state: Cx, cold_state: Cx }

## The header product a revision's own header passes wrote, captured and put
## back. What used to sit beside it -- the same tables projected onto a second
## revision -- is gone with `relocate_header`: a header's references are the
## same integers in both revisions, and its positions are offsets from the
## declaration that recorded them, so the projection had nothing left to do
## and the fixture had to invent the movement it was checking (K7).
pub(pkg) type MetadataTrial = { assembled: Cx, header_state: Cx }

pub(pkg) fn metadata_sample() -> MetadataTrial !io = {
  let text = "pub type Box[T] = { value: T }\n" ++
    "pub alias Thunk[T, !e] = fn(T) -> T !e\n" ++
    "pub opaque type Token = Int\n" ++
    "pub effect Ask { fn ask() -> Int }\n" ++
    "pub trait Carry[T] {\n type Item\n effect Run = !Ask\n fn get(x: T) -> T.Item !T.Run\n}\n" ++
    "impl[T] Carry[Box[T]] {\n type Item = T\n fn get(x: Box[T]) -> T !Ask = x.value\n}\n" ++
    "pub fn usebox[T](x: Box[T]) -> Box[T] = x\n" ++
    "pub const NAME: String = \"metadata\"\n"
  let prefix = "# moved header source\n"
  let edited = prefix ++ text
  let (syntax, ds) = parser.parse_module(text)
  let (next_syntax, nds) = parser.parse_module(edited)
  if ds != [] || nds != [] { panic("metadata parse") }
  let scope = module_key("metadata-world", "metadata", "metadata", "metadata.dawn")
  let initial = cx_new(Some("metadata"), Some("metadata.dawn"))
  let old = check_module_headers(initial, syntax, map.empty())
  let next = check_module_headers(initial, next_syntax, map.empty())
  if old.cx.diags != [] || next.cx.diags != [] { panic("metadata headers") }
  let reserved = allocation.compiler_headers(scope.world).expect("metadata compiler origins")
  let old_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, syntax, old.cx).expect("metadata old declarations")).expect("metadata old table"),
    allocation.table(allocation.local_impl_headers(scope, syntax, old.cx, old.impl_sigs).expect("metadata old impls")).expect("metadata old impl table")]).expect("metadata old combined")
  let next_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, next_syntax, next.cx).expect("metadata next declarations")).expect("metadata next table"),
    allocation.table(allocation.local_impl_headers(scope, next_syntax, next.cx, next.impl_sigs).expect("metadata next impls")).expect("metadata next impl table")]).expect("metadata next combined")
  if not allocation.between(old_table, next_table) { panic("metadata header join refused") }
  let old_exports = exports_of(old.cx, syntax, "metadata", "metadata")
  let cold = exports_of(next.cx, next_syntax, "metadata", "metadata")
  # A nominal id derives from the declaration, so shifting the counter under
  # the second revision leaves `Box` exactly where it was; what the projection
  # still has work to do on is the binder domain, and the two signatures below
  # are what prove this fixture reaches it.
  if map.get(old_exports.types_by_name, "Box") != map.get(cold.types_by_name, "Box") ||
    map.len(old_exports.aliases) != 2 || map.len(old_exports.effects) != 1 || len(old_exports.impls) == 0 {
    panic("metadata coverage")
  }
  # A prefix insertion used to renumber every binder in the module, and this
  # fixture required it to. A binder is a slot of its own declaration now, so
  # the two revisions export the same signatures and the projection's
  # remaining work here is the positional half below.
  if old_exports.fns != cold.fns { panic("metadata binder coverage") }
  let product = header_product.capture(initial, old.cx).expect("metadata header capture")
  let assembled = header_product.assemble(initial, product).expect("metadata header assembly")
  MetadataTrial { assembled: assembled, header_state: old.cx }
}

pub(pkg) fn module_samples() -> List[ModuleTrial] !io = [inferred_module_sample(), reordered_module_sample(), method_module_sample(), value_module_sample()] ++ default_module_samples()

type SavedFunction = { before: Cx, after: Cx, product: body_product.Product }
type SavedConstant = { before: Cx, after: Cx, product: body_product.ConstantProduct }

fn value_module_sample() -> ModuleTrial !io = {
  let first = "type First = { v: Int }\n"
  let second = "type Second = { v: Int }\n"
  let a = "const A: First = { let n = 1\n First { v: n } }\n"
  let b = "const B: Second = { let n = 2\n let m = n\n Second { v: m } }\n"
  let c = "const C: Int = A.v + B.v\n"
  let ta = "test \"A\" { let x = A.v\n assert x == 1 }\n"
  let tb = "test \"B\" { assert B.v == 2\n assert C == 3 }\n"
  let text = first ++ second ++ a ++ ta ++ b ++ c ++ tb
  let edited = "# moved values\n" ++ second ++ first ++ b ++ tb ++ a ++ c ++ ta
  let (syntax, ds) = parser.parse_module(text)
  let (next_syntax, nds) = parser.parse_module(edited)
  if ds != [] || nds != [] { panic("value module parse") }
  let scope = module_key("value-world", "values", "values", "values.dawn")
  let initial = cx_new(Some("values"), Some("values.dawn"))
  let old_headers = check_module_headers(initial, syntax, map.empty())
  let next_headers = check_module_headers(initial, next_syntax, map.empty())
  let old_named = function_product.values(scope, old_headers).expect("old value headers")
  let next_named = function_product.values(scope, next_headers).expect("new value headers")
  var constants: Map[DeclKey, SavedConstant] = map.empty()
  var tests: Map[DeclKey, SavedFunction] = map.empty()
  var old_cx = old_headers.cx
  for key in function_product.constant_keys(old_named) {
    match function_product.value(old_named, key).expect("old constant") {
      function_product.ConstantHeader(_, d, ty, visible) -> {
        let entered = owned_path(old_cx, syntax, key.path)
        let (after, tree) = check_const_init(entered, d, ty, visible)
        let product = body_product.capture_constant(entered, after, tree).expect("constant capture")
        constants = map.insert(constants, key, SavedConstant { before: entered, after: after, product: product })
        old_cx = unowned(after)
      }
      _ -> panic("constant role")
    }
  }
  for key in function_product.test_keys(old_named) {
    match function_product.value(old_named, key).expect("old test") {
      function_product.TestHeader(_, d) -> {
        let entered = owned_path(old_cx, syntax, key.path)
        let (after, tree) = check_test(entered, d.name, d.body)
        tests = map.insert(tests, key, SavedFunction { before: entered, after: after,
          product: body_product.capture(entered, after, tree).expect("test capture") })
        old_cx = unowned(after)
      }
      _ -> panic("test role")
    }
  }
  let reserved = allocation.compiler_headers(scope.world).expect("value compiler origins")
  let old_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, syntax, old_headers.cx).expect("old value origins")).expect("old value table")]).expect("old value combined")
  let next_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, next_syntax, next_headers.cx).expect("new value origins")).expect("new value table")]).expect("new value combined")
  if not allocation.between(old_table, next_table) { panic("value header join refused") }
  var replayed = next_headers.cx
  var moved_constants: List[TConst] = []
  var moved_tests: List[TFun] = []
  for key in function_product.constant_keys(next_named) {
    match function_product.value(next_named, key).expect("next constant") {
      function_product.ConstantHeader(_, d, ty, _) -> {
        let saved = map.get(constants, key).expect("saved constant")
        let p = saved.product
        if not allocation.constant_relocation(p.tree.ty, ty) { panic("constant allocation") }
        let view = View { symbols: saved.after.syms,
          callee: (owner, name, builtin, tr) => checked_callee(saved.before, owner, name, builtin, tr) }
        let projected = body_product.project_constant(view, p).expect("constant projection")
        let resolver = path_resolver(replayed, next_syntax, key.path)
        let installed = body_product.assemble(owned_path(replayed, next_syntax, key.path), projected)
          .expect("constant replay")
        replayed = left_decl(installed, resolver, mint_cursor(saved.before), mint_cursor(saved.after))
        moved_constants = moved_constants ++ [tast_positions.constant(resolver, projected.tree)]
      }
      _ -> panic("next constant role")
    }
  }
  for key in function_product.test_keys(next_named) {
    match function_product.value(next_named, key).expect("next test") {
      function_product.TestHeader(_, d) -> {
        let saved = map.get(tests, key).expect("saved test")
        let p = saved.product
        # Test signatures have only fixed Unit/io metadata, no header binders.
        let no_evidence: List[Int] = []
        if not allocation.body_relocation(p.tree.sig, p.tree.sig, no_evidence) { panic("test allocation") }
        let prior = match function_product.value(old_named, key).expect("prior test") {
          function_product.TestHeader(_, old) -> old
          _ -> panic("prior test role")
        }
        let view = View { symbols: saved.after.syms,
          callee: (owner, name, builtin, tr) => checked_callee(saved.before, owner, name, builtin, tr) }
        let projected = body_product.project(view, p).expect("test projection")
        let resolver = path_resolver(replayed, next_syntax, key.path)
        let installed = body_product.assemble(owned_path(replayed, next_syntax, key.path), projected)
          .expect("test replay")
        replayed = left_decl(installed, resolver, mint_cursor(saved.before), mint_cursor(saved.after))
        moved_tests = moved_tests ++ [tast_positions.function(resolver, projected.tree)]
      }
      _ -> panic("next test role")
    }
  }
  let (assembled_cx, assembled) = assemble_module_bodies(ModuleBodies { cx: replayed, functions: [],
    constants: moved_constants, tests: moved_tests, impl_methods: [], trait_defaults: [] })
  let (cold_cx, cold) = check_module(initial, next_syntax, map.empty())
  if old_cx.diags != [] || cold_cx.diags != [] { panic("value module diagnostics") }
  if len(assembled.consts) != 3 || len(assembled.tests) != 2 { panic("value module count") }
  ModuleTrial { replayed_cx: assembled_cx, cold_cx: cold_cx, replayed: assembled, cold: cold, states: [] }
}

fn method_signature(method: function_product.MethodHeader) -> Sig = match method {
  function_product.ImplMethod(_, _, sig, _) -> sig
  function_product.TraitDefault(_, _, sig, _) -> sig
}

fn method_span(method: function_product.MethodHeader) -> (Int, Int) = match method {
  function_product.ImplMethod(_, syntax, _, _) -> (syntax.lo, syntax.hi)
  function_product.TraitDefault(_, syntax, _, _) -> (syntax.lo, syntax.hi)
}

fn check_method_source(cx: Cx, method: function_product.MethodHeader) -> (Cx, TFun) !io = match method {
  function_product.ImplMethod(_, syntax, sig, _) -> check_fn(cx, strip_param_defaults(syntax), sig)
  function_product.TraitDefault(_, syntax, sig, _) -> check_trait_default(cx, syntax, sig, syntax.body.expect("default body"))
}

fn method_module_sample() -> ModuleTrial !io = {
  let forward = "trait Forward[T] {\n fn forward(x: T) -> Int\n fn alternate(x: T) -> Int\n fn twice(x: T) -> Int = forward(x) + alternate(x)\n}\n"
  let repeat = "trait Repeat[T] {\n fn repeat(x: T) -> Int\n fn other(x: T) -> Int\n fn again(x: T) -> Int = repeat(x) + other(x)\n}\n"
  let first = "impl[T] Forward[List[T]] {\n fn forward(x: List[T]) -> Int = 1\n fn alternate(x: List[T]) -> Int = 3\n}\n"
  let second = "impl[T] Repeat[List[T]] {\n fn repeat(x: List[T]) -> Int = 2\n fn other(x: List[T]) -> Int = 4\n}\n"
  let text = forward ++ repeat ++ first ++ second
  let edited = "# moved methods\n" ++ repeat ++ forward ++ second ++ first
  let (syntax, ds) = parser.parse_module(text)
  let (next_syntax, nds) = parser.parse_module(edited)
  if ds != [] || nds != [] { panic("method module parse") }
  let scope = module_key("method-world", "methods", "methods", "methods.dawn")
  let initial = cx_new(Some("methods"), Some("methods.dawn"))
  let old_headers = check_module_headers(initial, syntax, map.empty())
  let next_headers = check_module_headers(initial, next_syntax, map.empty())
  let old_methods = function_product.methods(scope, old_headers).expect("old method entries")
  let next_methods = function_product.methods(scope, next_headers).expect("next method entries")
  var saved: Map[DeclKey, SavedFunction] = map.empty()
  var old_cx = old_headers.cx
  for key in function_product.impl_keys(old_methods) ++ function_product.default_keys(old_methods) {
    let entry = function_product.method(old_methods, key).expect("old method")
    # `DeclKey.path` is the declaration this method is, which is what opens it.
    let entered = owned_path(old_cx, syntax, key.path)
    let (after, body) = check_method_source(entered, entry)
    if body.impl_of != None || body.default_of != None { panic("method body was already tagged") }
    saved = map.insert(saved, key, SavedFunction { before: entered, after: after,
      product: body_product.capture(entered, after, body).expect("method capture") })
    old_cx = unowned(after)
  }
  let reserved = allocation.compiler_headers(scope.world).expect("method compiler provenance")
  let old_entries = allocation.local_headers(scope, syntax, old_headers.cx).expect("old method headers") ++
    allocation.local_impl_headers(scope, syntax, old_headers.cx, old_headers.impl_sigs).expect("old impl headers")
  let next_entries = allocation.local_headers(scope, next_syntax, next_headers.cx).expect("next method headers") ++
    allocation.local_impl_headers(scope, next_syntax, next_headers.cx, next_headers.impl_sigs).expect("next impl headers")
  let old_table = allocation.combine([reserved, allocation.table(old_entries).expect("old method table")]).expect("old combined methods")
  let next_table = allocation.combine([reserved, allocation.table(next_entries).expect("next method table")]).expect("next combined methods")
  if not allocation.between(old_table, next_table) { panic("method header join refused") }
  var replayed = next_headers.cx
  var impls: List[TFun] = []
  var defaults: List[TFun] = []
  # The cold pipeline fixes the tag once per impl, not once per method.
  var tags: Map[DeclKey, Option[(Int, Ty)]] = map.empty()
  var order: List[DeclKey] = []
  for group in function_product.impl_groups(next_methods) {
    order = order ++ group.methods
  }
  order = order ++ function_product.default_keys(next_methods)
  for key in order {
    for group in function_product.impl_groups(next_methods) {
      if len(group.methods) > 0 && group.methods[0] == key {
        let tag = impl_body_key(replayed, group.trait_name, group.subject)
        for member in group.methods { tags = map.insert(tags, member, tag) }
      }
    }
    let entry = function_product.method(next_methods, key).expect("next method")
    let prior = function_product.method(old_methods, key).expect("prior method")
    let body = map.get(saved, key).expect("method body by key")
    if not allocation.body_relocation(method_signature(prior), method_signature(entry),
      body.product.tree.ev_syms) { panic("method allocation") }
    let (old_lo, old_hi) = method_span(prior)
    let (next_lo, next_hi) = method_span(entry)
    let view = View { symbols: body.after.syms,
      callee: (owner, name, builtin, tr) => checked_callee(body.before, owner, name, builtin, tr) }
    let projected = body_product.project(view, body.product).expect("method projection")
    let resolver = path_resolver(replayed, next_syntax, key.path)
    let installed = body_product.assemble(owned_path(replayed, next_syntax, key.path), projected)
      .expect("method replay")
    replayed = left_decl(installed, resolver, mint_cursor(body.before), mint_cursor(body.after))
    let moved_tree = tast_positions.function(resolver, projected.tree)
    match entry {
      function_product.ImplMethod(_, _, _, info) -> {
        impls = impls ++ [TFun { ..moved_tree, impl_of: map.get(tags, key).expect("current impl tag") }]
      }
      function_product.TraitDefault(_, _, _, tid) -> {
        defaults = defaults ++ [TFun { ..moved_tree, default_of: Some(tid) }]
      }
    }
  }
  if len(impls) != 4 || len(defaults) != 2 { panic("method products missing") }
  let (assembled_cx, assembled) = assemble_module_bodies(ModuleBodies {
    cx: replayed, functions: [], constants: [], impl_methods: impls, trait_defaults: defaults, tests: [] })
  let (cold_cx, cold) = check_module(initial, next_syntax, map.empty())
  if len(assembled_cx.diags) != 0 || len(cold_cx.diags) != 0 { panic("method module diagnostics") }
  ModuleTrial { replayed_cx: assembled_cx, cold_cx: cold_cx, replayed: assembled, cold: cold, states: [] }
}

fn reordered_module_sample() -> ModuleTrial !io = {
  let first = "fn first[T](x: T) -> T = { let a = x\n a }\n"
  let second = "fn second[U](x: U) -> U = { let b = x\n let c = b\n c }\n"
  let caller = "fn caller(x: Int) -> Int = first(x) + second(x)\n"
  let text = first ++ second ++ caller
  let edited = "# moved declarations\n" ++ second ++ first ++ caller
  let (syntax, ds) = parser.parse_module(text)
  let (next_syntax, nds) = parser.parse_module(edited)
  if ds != [] || nds != [] { panic("named module parse") }
  let scope = module_key("named-world", "named", "named", "named.dawn")
  let initial = cx_new(Some("named"), Some("named.dawn"))
  let old_headers = check_module_headers(initial, syntax, map.empty())
  let next_headers = check_module_headers(initial, next_syntax, map.empty())
  let old_named = function_product.headers(scope, old_headers).expect("old named headers")
  let next_named = function_product.headers(scope, next_headers).expect("next named headers")
  let old_keys = function_product.keys(old_named)
  let next_keys = function_product.keys(next_named)
  if old_keys[0] != next_keys[1] || old_keys[1] != next_keys[0] { panic("named module did not reorder") }
  # A declaration reorder used to renumber every generic binder after it.
  # Those binders are slots of their own declaration now, so the reordered
  # revision hands back the same signature, and this fixture says so rather
  # than requiring the shift it used to replay.
  if function_product.get(old_named, old_keys[0]).expect("old first").signature !=
      function_product.get(next_named, old_keys[0]).expect("next first").signature {
    panic("named module moved a generic header ID under a reorder")
  }
  var saved: Map[DeclKey, SavedFunction] = map.empty()
  var old_cx = old_headers.cx
  for key in old_keys {
    let header = function_product.get(old_named, key).expect("old declaration")
    let entered = owned_decl(old_cx, syntax, header.syntax)
    let (after, body) = check_fn(entered, header.syntax, header.signature)
    let product = body_product.capture(entered, after, body).expect("named capture")
    saved = map.insert(saved, key, SavedFunction { before: entered, after: after, product: product })
    old_cx = unowned(after)
  }
  let reserved = allocation.compiler_headers(scope.world).expect("named compiler origins")
  let old_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, syntax, old_headers.cx).expect("old named origins")).expect("old table")]).expect("old combined")
  let next_table = allocation.combine([reserved,
    allocation.table(allocation.local_headers(scope, next_syntax, next_headers.cx).expect("next named origins")).expect("next table")]).expect("next combined")
  if not allocation.between(old_table, next_table) { panic("named header join refused") }
  var replayed = next_headers.cx
  var functions: List[TFun] = []
  for key in next_keys {
    let entry = function_product.get(next_named, key).expect("next declaration")
    let prior = function_product.get(old_named, key).expect("prior declaration")
    let body = map.get(saved, key).expect("named body lookup")
    if not allocation.body_relocation(prior.signature, entry.signature,
      body.product.tree.ev_syms) { panic("named allocation") }
    let view = View { symbols: body.after.syms,
      callee: (owner, name, builtin, tr) => checked_callee(body.before, owner, name, builtin, tr) }
    let projected = body_product.project(view, body.product).expect("named projection")
    let resolver = path_resolver(replayed, next_syntax, [Named(FunctionDecl, entry.syntax.name)])
    let installed = body_product.assemble(owned_decl(replayed, next_syntax, entry.syntax), projected)
      .expect("named replay")
    replayed = left_decl(installed, resolver, mint_cursor(body.before), mint_cursor(body.after))
    functions = functions ++ [tast_positions.function(resolver, projected.tree)]
  }
  let (assembled_cx, assembled) = assemble_module_bodies(ModuleBodies {
    cx: replayed, functions: functions, constants: [], impl_methods: [], trait_defaults: [], tests: [] })
  let (cold_cx, cold) = check_module(initial, next_syntax, map.empty())
  ModuleTrial { replayed_cx: assembled_cx, cold_cx: cold_cx, replayed: assembled, cold: cold, states: [] }
}

pub(pkg) fn import_samples() -> List[StateTrial] !io = [import_sample(0), import_sample(1), import_sample(2)]

## Three consumer shapes over the same reordered provider. 0 imports the names
## themselves; 1 keeps the module name and spells a qualified type in the
## signature plus a qualified call; 2 spells a qualified type in a body
## annotation and reads a qualified constant. The current effect syntax does
## not accept `!dep.Ask`, so there is no qualified effect case to write.
fn import_sample(kind: Int) -> StateTrial !io = {
  let qualified = kind != 0
  let ask = if qualified { "pub type First = { value: Int }\n" } else { "pub effect Ask { fn ask() -> Int }\n" }
  let tell = if qualified { "pub type Second = { value: Int }\n" } else { "pub effect Tell { fn tell() -> Int }\n" }
  let tail = if kind == 1 { "pub fn identity[T](x: T) -> T = x\n" }
    else if kind == 2 { "pub const ORIGIN: First = First { value: 0 }\n" }
    else { "" }
  let (old_library, old_pd) = parser.parse_module(ask ++ tell ++ tail)
  let (new_library, new_pd) = parser.parse_module(tell ++ ask ++ tail)
  let base = cx_new(Some("library"), Some("library.dawn"))
  let (provider, _) = check_module(base, old_library, map.empty())
  let (new_provider, _) = check_module(base, new_library, map.empty())
  if len(old_pd) != 0 || len(new_pd) != 0 || len(provider.diags) != 0 || len(new_provider.diags) != 0 {
    panic("import provider fixture failed")
  }
  let old_env = map.from([("library", exports_of(provider, old_library, "library", "library"))])
  let new_env = map.from([("library", exports_of(new_provider, new_library, "library", "library"))])
  let text = if kind == 1 { "use library as dep\npub fn both(x: dep.First) -> dep.First = dep.identity(x)\n" }
    else if kind == 2 { "use library as dep\npub fn both() -> Int = { let v: dep.First = dep.ORIGIN\n v.value }\n" }
    else { "use library.{Ask, Tell}\npub fn both() -> Int !Ask !Tell = ask() + tell()\n" }
  let (syntax, pd) = parser.parse_module(text)
  if len(pd) != 0 { panic("import consumer parse failed: " ++ join(map(pd, d => d.msg), "; ")) }
  let consumer = cx_new(Some("consumer"), Some("consumer.dawn"))
  let old_base = consumer
  let new_base = consumer
  let (before, sigs) = headers_for_body_probe(old_base, syntax, old_env)
  let (next, nsigs) = headers_for_body_probe(new_base, syntax, new_env)
  if len(before.diags) != 0 || len(next.diags) != 0 { panic("import consumer headers failed") }
  let library_scope = module_key("import-world", "library", "library", "library.dawn")
  let scope = module_key("import-world", "consumer", "consumer", "consumer.dawn")
  let old_library_table = allocation.table(allocation.local_headers(library_scope, old_library, provider).expect("old provider provenance")).expect("old provider table")
  let new_library_table = allocation.table(allocation.local_headers(library_scope, new_library, new_provider).expect("new provider provenance")).expect("new provider table")
  let old_local = allocation.table(allocation.local_headers(scope, syntax, before).expect("old consumer provenance")).expect("old consumer table")
  let new_local = allocation.table(allocation.local_headers(scope, syntax, next).expect("new consumer provenance")).expect("new consumer table")
  let reserved = allocation.compiler_headers("import-world").expect("intrinsics")
  let old_table = allocation.combine([old_library_table, old_local, reserved]).expect("old program provenance")
  let new_table = allocation.combine([new_library_table, new_local, reserved]).expect("new program provenance")
  let old_ask = if qualified { map.get(provider.adts_by_name, "First").expect("old exported type") }
    else { map.get(provider.effects, "Ask").expect("old exported effect") }
  let new_ask = if qualified { map.get(new_provider.adts_by_name, "First").expect("new exported type") }
    else { map.get(new_provider.effects, "Ask").expect("new exported effect") }
  # A derived id: the provider declares the same thing in both revisions, so
  # the reference the consumer holds is the same integer and no ledger entry
  # has to say so. The join still has to exist for the binder domains.
  if old_ask != new_ask { panic("provider identity moved under a reorder") }
  if not allocation.between(old_table, new_table) { panic("program join refused") }
  if not allocation.between(old_local, new_local) { panic("consumer join refused") }
  let decl = module_fns(syntax)[0]
  let entered_old = owned_decl(before, syntax, decl)
  let entered_new = owned_decl(next, syntax, decl)
  let resolver = path_resolver(next, syntax, [Named(FunctionDecl, decl.name)])
  let (after, body) = check_fn(entered_old, decl, sigs[0])
  if not allocation.between(old_table, new_table) { panic("import header join refused") }
  if not allocation.body_relocation(body.sig, nsigs[0], body.ev_syms) { panic("import body plan") }
  let view = View { symbols: after.syms,
    callee: (owner, name, builtin, tr) => checked_callee(entered_old, owner, name, builtin, tr) }
  let saved = body_product.capture(entered_old, after, body).expect("import body capture")
  let projected = body_product.project(view, saved).expect("import body projection")
  let replayed = left_decl(body_product.assemble(entered_new, projected).expect("import body assembly"),
    resolver, mint_cursor(entered_old), mint_cursor(after))
  let (cold_inside, raw_cold) = check_fn(entered_new, decl, nsigs[0])
  let cold = left_decl(cold_inside, resolver, mint_cursor(entered_new), mint_cursor(cold_inside))
  let cold_body = tast_positions.function(resolver, raw_cold)
  let (_, whole) = check_module(new_base, syntax, new_env)
  if len(after.diags) != 0 || len(cold.diags) != 0 { panic("import consumer body failed") }
  StateTrial { replayed: replayed, cold: cold, relocated: tast_positions.function(resolver, projected.tree),
    cold_body: cold_body, module_body: whole.fns[0] }
}

pub(pkg) fn default_samples() -> List[StateTrial] !io = map(default_module_samples(), m => m.states[0])

fn default_module_samples() -> List[ModuleTrial] !io = [default_sample(false, false, false), default_sample(true, false, false),
  default_sample(false, true, false), default_sample(true, true, false),
  default_sample(false, false, true), default_sample(true, false, true)]

fn default_sample(inferred: Bool, generic: Bool, invalid: Bool) -> ModuleTrial !io = {
  let prefix = if generic { "trait Marker[T] { fn mark(x: T) -> Int }\nfn sample[T: Marker]" } else { "fn sample" }
  let params = if generic { "(x: fn(T) -> Int = (t) => mark(t), y: Int = { let w = 2\n w })" }
    else if invalid { "(x: Int = true, y: Int = { let w = 2\n w })" }
    else { "(x: Int = { let v = 1\n v }, y: Int = { let w = 2\n w })" }
  let text = prefix ++ params ++ (if inferred { " = " } else { " -> Int = " }) ++
    (if generic { "y\n" } else { "x + y\n" })
  let (syntax, diags) = parser.parse_module(text)
  let edited = "# 🌅 moved defaults\n" ++ text
  let (new_syntax, new_diags) = parser.parse_module(edited)
  if len(new_diags) != 0 { panic("edited default boundary parse failed") }
  if len(diags) != 0 { panic("default boundary parse failed") }
  let base = cx_new(Some("defaults"), Some("defaults.dawn"))
  let new_base = base
  let (before, sigs) = headers_for_body_probe(base, syntax, map.empty())
  let (next, nsigs) = headers_for_body_probe(new_base, new_syntax, map.empty())
  let decl = module_fns(syntax)[0]
  let new_decl = module_fns(new_syntax)[0]
  let scope = module_key("defaults-world", "defaults", "defaults", "defaults.dawn")
  let old_table = allocation.table(allocation.local_headers(scope, syntax, before).expect("old default headers")).expect("old table")
  let new_table = allocation.table(allocation.local_headers(scope, new_syntax, next).expect("new default headers")).expect("new table")
  # The defaults and the body that follows them are separate boundaries of one
  # declaration. Each was an interval the next one's relocation had to carry;
  # they are runs of the declaration's own slots now, so each boundary is
  # admitted on its own header and nothing accumulates.
  if not allocation.between(old_table, new_table) { panic("default header join refused") }
  # One declaration, two boundaries: the parameter defaults and the body that
  # follows them are checked inside the same open declaration, so the fixture
  # opens it once and leaves it once, the way the scheduler does.
  let resolver = path_resolver(next, new_syntax, [Named(FunctionDecl, new_decl.name)])
  let entered_old = owned_decl(before, syntax, decl)
  let entered_new = owned_decl(next, new_syntax, new_decl)
  var old_cx = entered_old
  var new_cx = entered_new
  var old_defaults: List[Option[TDefault]] = []
  var k = 0
  for p in decl.params {
    let (after, value) = check_param_default(old_cx, decl, sigs[0], k)
    if len(value.dict_syms) != (if generic { 1 } else { 0 }) { panic("default dictionary fixture missing its bound") }
    let sig = default_sig(sigs[0], decl.name, k, sigs[0].param_tys[k])
    let new_sig = default_sig(nsigs[0], decl.name, k, nsigs[0].param_tys[k])
    let fn_value = TFun { name: sig.name, sig: sig, param_syms: [], dict_syms: value.dict_syms,
      defaults: [], ev_syms: [], body: value.body, is_test: false, impl_of: None, default_of: None }
    let no_evidence: List[Int] = []
    if not allocation.body_relocation(sig, new_sig, no_evidence) { panic("default allocation") }
    let view = View { symbols: after.syms,
      callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
    let saved = body_product.capture(old_cx, after, fn_value).expect("default capture")
    let projected = body_product.project(view, saved).expect("default projection")
    let default_installed = body_product.assemble(new_cx, projected).expect("default assembly")
    new_cx = default_installed
    old_defaults = old_defaults ++ [Some(value)]
    old_cx = after
    k = k + 1
  }
  let (old_after, old_body) = if inferred {
    let (cx, _, value) = check_fn_inferred_body(old_cx, decl, sigs[0], old_defaults)
    (cx, value)
  } else { check_fn_body(old_cx, decl, sigs[0], old_defaults) }
  if not allocation.body_segment_relocation(sigs[0], nsigs[0], old_body.ev_syms) { panic("main allocation") }
  let main_view = View { symbols: old_after.syms,
    callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
  let main_saved = body_product.capture(old_cx, old_after, old_body).expect("main capture")
  let main_projected = body_product.project(main_view, main_saved).expect("main projection")
  let main_installed = body_product.assemble(new_cx, main_projected).expect("main assembly")
  let moved_by = new_decl.lo - decl.lo
  let replayed = shifted_diags(
    left_decl(main_installed, resolver, mint_cursor(entered_old), mint_cursor(old_after)),
    len(next.diags), moved_by)
  let body = tast_positions.function(resolver, main_projected.tree)
  let (cold_inside, raw_cold) = if inferred {
    let (cx, _, value) = check_fn_inferred(entered_new, new_decl, nsigs[0])
    (cx, value)
  } else { check_fn(entered_new, new_decl, nsigs[0]) }
  let cold = left_decl(cold_inside, resolver, mint_cursor(entered_new), mint_cursor(cold_inside))
  let cold_body = tast_positions.function(resolver, raw_cold)
  let (whole_cx, whole) = check_module(new_base, new_syntax, map.empty())
  let (assembled_cx, assembled) = assemble_module_bodies(ModuleBodies {
    cx: replayed, functions: [body], constants: [], impl_methods: [], trait_defaults: [], tests: [] })
  let expected_diags = if invalid { 1 } else { 0 }
  if len(replayed.diags) != expected_diags || len(cold.diags) != expected_diags { panic("default boundary check failed") }
  ModuleTrial { replayed_cx: assembled_cx, cold_cx: whole_cx, replayed: assembled, cold: whole,
    states: [StateTrial { replayed: replayed, cold: cold, relocated: body, cold_body: cold_body, module_body: whole.fns[0] }] }
}

## The source order here is deliberately dependency order. This checks state
## products, not a replacement scheduler: the module oracle still uses the
## production inferred-function scheduling loop independently.
pub(pkg) fn inferred_samples() -> List[StateTrial] !io = inferred_module_sample().states

fn inferred_module_sample() -> ModuleTrial !io = {
  let text = "alias Counter = Int\nfn lead(x: Int) = { let y = x + 1\n y }\n" ++
    "fn inferred(x: Counter) = { let z = x + 2\n z }\n" ++
    "fn caller(x: Int) = inferred(x)\n" ++
    "fn maker[T](x: T) = (y: T) => x\n" ++
    "fn consumer(x: Int) = maker(x)\n" ++
    "fn effect_maker[T, !e](x: T, f: fn(T) -> T !e) = (y: T) => f(x)\n" ++
    "fn effect_consumer(x: Int) = effect_maker(x, (z: Int) => z)\n"
  let edited = "# 🌅 inferred edit\n" ++ str.replace(text, "let y = x + 1", "let extra = x + 1\n let y = extra")
  let (old_ast, old_diags) = parser.parse_module(text)
  let (new_ast, new_diags) = parser.parse_module(edited)
  if len(old_diags) != 0 || len(new_diags) != 0 { panic("inferred fixture parse") }
  let initial = cx_new(Some("inferred_probe"), Some("inferred.dawn"))
  let (old_header, old_sigs) = headers_for_body_probe(initial, old_ast, map.empty())
  let (new_header, new_sigs) = headers_for_body_probe(initial, new_ast, map.empty())
  let (whole_cx, whole) = check_module(initial, new_ast, map.empty())
  if len(old_header.diags) != 0 || len(new_header.diags) != 0 || len(whole_cx.diags) != 0 {
    panic("inferred fixture diagnostics")
  }
  let old_decls = module_fns(old_ast)
  let new_decls = module_fns(new_ast)
  let scope = module_key("inferred-world", "inferred_probe", "inferred_probe", "inferred.dawn")
  var old_table = allocation.table(allocation.local_headers(scope, old_ast, old_header).expect("old inferred headers")).expect("old inferred table")
  var new_table = allocation.table(allocation.local_headers(scope, new_ast, new_header).expect("new inferred headers")).expect("new inferred table")
  let reserved = allocation.compiler_headers("inferred-world").expect("inferred intrinsics")
  old_table = allocation.combine([old_table, reserved]).expect("old inferred provenance")
  new_table = allocation.combine([new_table, reserved]).expect("new inferred provenance")
  if not allocation.between(old_table, new_table) { panic("inferred header join refused") }
  var old = old_header
  var cold = new_header
  var replayed = new_header
  var out: List[StateTrial] = []
  var functions: List[TFun] = []
  var i = 0
  while i < len(old_decls) {
    if not old_sigs[i].inferring || not new_sigs[i].inferring { panic("fixture signature was not inferred") }
    let before = owned_decl(old, old_ast, old_decls[i])
    let (after, sealed, body) = check_fn_inferred(before, old_decls[i], old_sigs[i])
    let entered_cold = owned_decl(cold, new_ast, new_decls[i])
    let resolver = decl_resolver(cold, new_ast, new_decls[i])
    let (checked_inside, _, raw_cold) =
      check_fn_inferred(entered_cold, new_decls[i], new_sigs[i])
    let checked = left_decl(checked_inside, resolver, mint_cursor(entered_cold), mint_cursor(checked_inside))
    let cold_body = tast_positions.function(resolver, raw_cold)
    if sealed.inferring || map.get(after.fns, old_decls[i].name) != Some(sealed) { panic("inferred signature not sealed") }
    if i == 0 {
      replayed = checked
      functions = functions ++ [cold_body]
    } else {
      let delta = 0
      if not allocation.body_relocation(old_sigs[i], new_sigs[i], body.ev_syms) { panic("inferred allocation") }
      let view = View { symbols: after.syms,
        callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
      let saved = body_product.capture(before, after, body).expect("inferred capture")
      let projected = body_product.project(view, saved).expect("inferred projection: " ++ body.name)
      let installed = body_product.assemble(replayed, projected).expect("inferred assembly")
      # Nothing here is shifted by a constant any more. A product records
      # offsets into its own declaration, so putting it into the candidate
      # revision is resolving it against that declaration's position there,
      # which is what the scheduler does on the way out.
      replayed = left_decl(installed, resolver, mint_cursor(before), mint_cursor(after))
      let moved = tast_positions.function(resolver, projected.tree)
      functions = functions ++ [moved]
      out = out ++ [StateTrial { replayed: replayed, cold: checked, relocated: moved,
        cold_body: cold_body, module_body: whole.fns[i] }]
    }
    old = unowned(after)
    cold = checked
    i = i + 1
  }
  let (assembled_cx, assembled) = assemble_module_bodies(ModuleBodies {
    cx: replayed, functions: functions, constants: [], impl_methods: [], trait_defaults: [], tests: [] })
  ModuleTrial { replayed_cx: assembled_cx, cold_cx: whole_cx, replayed: assembled, cold: whole, states: out }
}

pub(pkg) fn test_samples() -> List[StateTrial] !io = {
  let text = "test \"lead\" { let x = 1\n assert x == 1 }\n" ++
    "test \"body\" { let y = 2\n assert y > 0 }\n"
  # The edit lengthens the first test and moves the second; the second's own
  # text is untouched, which is the only shape a saved body is admitted under.
  let edited = "# 🌅 test edit\n" ++ str.replace(text, "let x = 1", "let extra = 1\n let x = extra")
  let (old_ast, old_diags) = parser.parse_module(text)
  let (new_ast, new_diags) = parser.parse_module(edited)
  if len(old_diags) != 0 || len(new_diags) != 0 { panic("test fixture parse") }
  let initial = cx_new(Some("test_probe"), Some("test.dawn"))
  let (old_header, _) = headers_for_body_probe(initial, old_ast, map.empty())
  let (new_header, _) = headers_for_body_probe(initial, new_ast, map.empty())
  let (whole_cx, whole) = check_module(initial, new_ast, map.empty())
  if len(whole_cx.diags) != 0 { panic("test fixture diagnostics") }
  let old_tests = module_tests(old_ast)
  let new_tests = module_tests(new_ast)
  let lead_old = owned_path(old_header, old_ast, [Named(TestDecl, old_tests[0].name)])
  let (lead_after, _) = check_test(lead_old, old_tests[0].name, old_tests[0].body)
  let before = left_decl(lead_after, path_resolver(old_header, old_ast, [Named(TestDecl, old_tests[0].name)]),
    mint_cursor(lead_old), mint_cursor(lead_after))
  let lead_new = owned_path(new_header, new_ast, [Named(TestDecl, new_tests[0].name)])
  let (lead_current, _) = check_test(lead_new, new_tests[0].name, new_tests[0].body)
  let current = left_decl(lead_current, path_resolver(new_header, new_ast, [Named(TestDecl, new_tests[0].name)]),
    mint_cursor(lead_new), mint_cursor(lead_current))
  let d = old_tests[1]
  let n = new_tests[1]
  let entered_old = owned_path(before, old_ast, [Named(TestDecl, d.name)])
  let (after, body) = check_test(entered_old, d.name, d.body)
  let entered_new = owned_path(current, new_ast, [Named(TestDecl, n.name)])
  let resolver = path_resolver(current, new_ast, [Named(TestDecl, n.name)])
  let (cold_inside, raw_cold) = check_test(entered_new, n.name, n.body)
  let cold = left_decl(cold_inside, resolver, mint_cursor(entered_new), mint_cursor(cold_inside))
  let cold_body = tast_positions.function(resolver, raw_cold)
  let delta = 0
  # Assembled into the declaration the scheduler would have opened, not into
  # the context outside it: a test block is not visited by the header passes,
  # so opening it is where its own identity is minted.
  let replayed = left_decl(source_state(entered_new, entered_old, after, body, delta), resolver,
    mint_cursor(entered_old), mint_cursor(after))
  let moved = tast_positions.function(resolver,
    admitted_body(local_view(entered_old, after, delta), body))
  [StateTrial { replayed: replayed, cold: cold, relocated: moved, cold_body: cold_body, module_body: whole.tests[1] }]
}

## Reverse real effect declarations, so label IDs and ABI slots reverse.
## Target body-local slots are derived from the new header's binding order,
## not read out of the cold body used as the comparison oracle.
pub(pkg) fn header_sample() -> HeaderTrial !io = {
  let ask = "pub effect Ask { fn ask() -> Int }\n"
  let tell = "pub effect Tell { fn tell() -> Int }\n"
  let first_type = "type First[T, !e] = { f: fn(T) -> T !e }\n"
  let second_type = "type Second[T, !e] = { value: T }\n"
  let first_alias = "opaque type Wrapped[T, !e] = fn(T) -> T !e\n"
  let second_alias = "alias Callback[T, !e] = fn(T) -> T !e\n"
  let first_trait = "trait Forward[T] {\n fn forward(x: T, f: fn() -> Int !e) -> Int !e\n fn forward_twice(x: T, f: fn() -> Int !e) -> Int !e = forward(x, f) + forward(x, f)\n}\n"
  let second_trait = "trait Repeat[T] {\n fn repeat(x: T, f: fn() -> Int !e) -> Int !e\n fn repeat_twice(x: T, f: fn() -> Int !e) -> Int !e = repeat(x, f) + repeat(x, f)\n}\n"
  let first_impl = "impl[T] Forward[List[T]] { fn forward(x: List[T], f: fn() -> Int !e) -> Int !e = f() }\n"
  let second_impl = "impl[T] Repeat[List[T]] { fn repeat(x: List[T], f: fn() -> Int !e) -> Int !e = f() }\n"
  let tail = "pub fn thru[!e](g: fn() -> Int !e) -> Int !e = g()\n" ++
    "pub fn both() -> Int !Ask !Tell = thru(() => ask() + tell())\n"
  let old_text = first_type ++ second_type ++ first_alias ++ second_alias ++ first_trait ++ second_trait ++ first_impl ++ second_impl ++ ask ++ tell ++ tail
  let new_text = second_type ++ first_type ++ second_alias ++ first_alias ++ second_trait ++ first_trait ++ second_impl ++ first_impl ++ tell ++ ask ++ tail
  let (old_ast, opd) = parser.parse_module(old_text)
  let (new_ast, npd) = parser.parse_module(new_text)
  if len(opd) != 0 || len(npd) != 0 { panic("header fixture parse failed") }
  let old_headers = check_module_headers(cx_new(Some("header"), Some("header.dawn")), old_ast, map.empty())
  let new_headers = check_module_headers(cx_new(Some("header"), Some("header.dawn")), new_ast, map.empty())
  let before = old_headers.cx
  let next = new_headers.cx
  let sigs = old_headers.sigs
  let nsigs = new_headers.sigs
  let impl_sigs = old_headers.impl_sigs
  let new_impl_sigs = new_headers.impl_sigs
  if len(before.diags) != 0 || len(next.diags) != 0 { panic("header fixture invalid") }
  let decl = module_fns(old_ast)[1]
  let ndecl = module_fns(new_ast)[1]
  let (after, saved) = check_fn(before, decl, sigs[1])
  let (checked, cold) = check_fn(next, ndecl, nsigs[1])
  if len(after.diags) != 0 || len(checked.diags) != 0 { panic("header body failed") }
  let old_ask = map.get(before.effects, "Ask").expect("old Ask")
  let old_tell = map.get(before.effects, "Tell").expect("old Tell")
  let new_ask = map.get(next.effects, "Ask").expect("new Ask")
  let new_tell = map.get(next.effects, "Tell").expect("new Tell")
  # Reordering the two effect declarations leaves both of their ids alone.
  if old_ask != new_ask || old_tell != new_tell || old_ask == old_tell { panic("effect identity moved under a reorder") }
  let scope = module_key("header-world", "header", "header", "header.dawn")
  var old_entries = allocation.local_headers(scope, old_ast, before).expect("old header provenance")
  var new_entries = allocation.local_headers(scope, new_ast, next).expect("new header provenance")
  old_entries = old_entries ++ allocation.local_impl_headers(scope, old_ast, before, impl_sigs).expect("old impl provenance")
  new_entries = new_entries ++ allocation.local_impl_headers(scope, new_ast, next, new_impl_sigs).expect("new impl provenance")
  if allocation.local_headers(ModuleKey { ..scope, source: "foreign.dawn" }, old_ast, before) != None {
    panic("foreign source accepted for local headers")
  }
  if allocation.local_impl_headers(scope, old_ast, before, []) != None {
    panic("missing impl signature table accepted")
  }
  if allocation.local_impl_headers(scope, old_ast, before, [impl_sigs[1], impl_sigs[0]]) != None {
    panic("reordered impl signature table accepted")
  }
  if allocation.local_impl_headers(scope, old_ast, Cx { ..before, local_impls: [] }, impl_sigs) != None {
    panic("missing impl provenance accepted")
  }
  let original_impl_sig = impl_sigs[0][0].expect("impl signature guard fixture")
  for invalid in [Sig { ..original_impl_sig, owner: Some("foreign") },
    Sig { ..original_impl_sig, tparams: [] }, Sig { ..original_impl_sig, is_builtin: true },
    Sig { ..original_impl_sig, trait_id: Some(999) }, Sig { ..original_impl_sig, op_of: Some((999, 0)) }] {
    if allocation.local_impl_headers(scope, old_ast, before, [[Some(invalid)], impl_sigs[1]]) != None {
      panic("invalid impl method metadata accepted")
    }
  }
  let reserved = allocation.compiler_headers("header-world").expect("header intrinsics")
  let old_table = allocation.combine([allocation.table(old_entries).expect("old local table"), reserved]).expect("old header table")
  let new_table = allocation.combine([allocation.table(new_entries).expect("new local table"), reserved]).expect("new header table")
  if not allocation.between(old_table, new_table) { panic("header body join refused") }
  if not allocation.body_relocation(saved.sig, nsigs[1], saved.ev_syms) { panic("body allocation plan") }
  let old_methods = function_product.methods(scope, old_headers).expect("old named methods")
  let new_methods = function_product.methods(scope, new_headers).expect("new named methods")
  if len(function_product.impl_keys(old_methods)) != 2 || len(function_product.default_keys(old_methods)) != 2 {
    panic("method header roles missing")
  }
  for key in function_product.impl_keys(new_methods) ++ function_product.default_keys(new_methods) {
    let old_method = function_product.method(old_methods, key).expect("old method by key")
    let new_method = function_product.method(new_methods, key).expect("new method by key")
    match (old_method, new_method) {
      (function_product.ImplMethod(_, _, old_sig, old_info), function_product.ImplMethod(_, _, new_sig, new_info)) -> {
        if old_sig != new_sig || old_info.subject != new_info.subject ||
            old_info.trait_id != new_info.trait_id { panic("named impl header projection") }
      }
      (function_product.TraitDefault(_, _, old_sig, old_tid), function_product.TraitDefault(_, _, new_sig, new_tid)) -> {
        if old_sig != new_sig || old_tid != new_tid {
          panic("named default header projection")
        }
      }
      _ -> panic("named method role mismatch")
    }
  }
  if slots_of(checked, after.owner_decl) != slots_of(after, after.owner_decl) { panic("body allocation endpoint differs from cold") }
  for name in ["First", "Second"] {
    let old_id = map.get(before.adts_by_name, name).expect("old ADT ID")
    let new_id = map.get(next.adts_by_name, name).expect("new ADT ID")
    let old_adt = map.get(before.adts, old_id).expect("old ADT")
    let new_adt = map.get(next.adts, new_id).expect("new ADT")
    if old_id != new_id ||
      old_adt.tparams[0] != new_adt.tparams[0] ||
      old_adt.bound_eparams[0] != new_adt.bound_eparams[0] {
      panic("ADT header provenance mismatch")
    }
  }
  for name in ["Wrapped", "Callback"] {
    let old_alias = map.get(before.aliases, name).expect("old alias")
    let new_alias = map.get(next.aliases, name).expect("new alias")
    if old_alias.tparams[0] != new_alias.tparams[0] ||
      old_alias.eparams[0] != new_alias.eparams[0] {
      panic("alias binder provenance mismatch")
    }
    if old_alias.is_opaque && old_alias.id != new_alias.id {
      panic("opaque alias provenance mismatch")
    }
  }
  for name in ["Forward", "Repeat"] {
    let old_id = map.get(before.traits_by_name, name).expect("old trait ID")
    let new_id = map.get(next.traits_by_name, name).expect("new trait ID")
    let old_trait = map.get(before.traits, old_id).expect("old trait")
    let new_trait = map.get(next.traits, new_id).expect("new trait")
    let (_, old_method) = old_trait.methods[0]
    let (_, new_method) = new_trait.methods[0]
    if old_id != new_id ||
      old_trait.tvar != new_trait.tvar ||
      old_method.sig != new_method.sig {
      panic("trait header provenance mismatch")
    }
  }
  var impl_index = 0
  while impl_index < len(impl_sigs) {
    let old_sig = impl_sigs[impl_index][0].expect("old impl method")
    let new_sig = new_impl_sigs[1 - impl_index][0].expect("new impl method")
    if old_sig != new_sig { panic("impl method provenance mismatch") }
    impl_index = impl_index + 1
  }
  let v = View { symbols: after.syms,
    callee: (owner, name, builtin, tr) => checked_callee(before, owner, name, builtin, tr) }
  let product = body_product.capture(before, after, saved).expect("header body capture")
  let projected = body_product.project(v, product).expect("header body projection")
  let moved_by = ndecl.lo - decl.lo
  let installed = body_product.assemble(next, projected).expect("header body assembly")
  HeaderTrial { relocated: shifted(moved_by, str.len(old_text), projected.tree), cold: cold,
    replayed_state: shifted_syms(next, installed, moved_by, str.len(old_text)), cold_state: checked }
}
