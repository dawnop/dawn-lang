# Frozen ordinary body loop from f688f4b4, before function_entry extraction.
# Appended only to a private checker subject; private entry/exit helpers remain
# shared so the oracle isolates this extraction rather than cloning semantics.
fn entry_frozen_body(cx: Cx, d: FnDecl, s: Sig, dtxs: List[Option[TDefault]]) -> (Cx, TFun) !io = {
  var cx1 = enter_fn(cx, s, d.nlo, d.nhi)
  var psyms: List[Int] = []
  var i = 0
  for p in d.params {
    let (cxa, sid) = declare(cx1, p.name, s.param_tys[i], false, p.lo, p.hi)
    cx1 = cxa
    psyms = psyms ++ [sid]
    i = i + 1
  }
  let (cx2, bt, bx) = check_expr(cx1, d.body, Some(s.ret))
  cx1 = cx2
  let (compatible_cx, compatible) = assignable_read(cx1, bt, s.ret)
  cx1 = compatible_cx
  if not compatible {
    let (return_cx, return_text) = type_display_read(cx1, s.ret)
    let (body_cx, body_text) = type_display_read(return_cx, bt)
    cx1 = cerr(body_cx, "function `" ++ d.name ++ "` declares return type " ++
        return_text ++ " but its body is " ++ body_text,
      e_lo(d.body), e_hi(d.body))
  }
  cx1 = verify_effects(cx1, "function", d.name, s.eff, s.eparams, "the end of the signature",
    d.nlo, d.nhi)
  let dsyms = dict_sym_list(cx1, s)
  let (cxes, esyms) = ev_sym_list(cx1, s)
  cx1 = cxes
  cx1 = pop_scope(cx1)
  (cx1, TFun {
    name: d.name,
    sig: s,
    param_syms: psyms,
    dict_syms: dsyms,
    ev_syms: esyms,
    defaults: dtxs,
    body: bx,
    is_test: false,
    impl_of: None,
    default_of: None,
    decl: ""
  })
}

pub(pkg) fn entry_frozen_fn(cx: Cx, d: FnDecl, s: Sig) -> (Cx, TFun) !io = {
  let (cx0, defaults) = check_param_defaults(cx, d, s)
  entry_frozen_body(cx0, d, s, defaults)
}

pub(pkg) fn entry_after_defaults(cx: Cx, d: FnDecl, s: Sig) -> FunctionEntry !io = {
  let (cx0, _) = check_param_defaults(cx, d, s)
  function_entry(cx0, d, s)
}
