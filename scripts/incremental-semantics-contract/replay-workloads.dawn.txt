# Synthetic replay workloads and the cold-equality check, shared by
# bench-replay and edit-matrix.
#
# The two harnesses must mean the same module when they say "calls, n=1000",
# and the same thing when they say "replay equals cold": the edit matrix
# reports hit rates for the Scale1000 subjects bench-replay times, and a second
# copy of either would be free to drift from the first without either harness
# noticing. The template mechanism copies one file per module, so each harness
# copies this file next to its own subject as `src/workloads.dawn`.
#
# A class is a fixed support prefix plus n members, one member per index. The
# member's body literal is a parameter rather than always its index so that an
# edit can change exactly one body (a different literal, or `false` for a type
# error) while keeping its name and every other byte of the module.
use check/checker.{ModuleBodies}

fn join(parts: List[String]) -> String = {
  var out = ""
  for p in parts { out = out ++ p ++ "\n" }
  out
}

## The declarations every member of a class may call. O(1) in n.
pub(pkg) fn support(kind: String) -> List[String] = {
  if kind == "calls" {
    return ["fn helper(x: Int) -> Int = x + 1", "fn helper2(x: Int, y: Int) -> Int = x * y - 1"]
  }
  if kind == "primitive_inferred" { return ["fn seed() = 1"] }
  if kind == "generic" {
    return ["trait Scale[T] { fn scale(x: T, k: Int) -> Int }",
      "type Coin = { value: Int }",
      "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Int = x.value * k }"]
  }
  if kind == "method" { return ["trait Shine[T] { fn shine(x: T) -> Int }"] }
  if kind == "java" { return ["use java \"java.lang.Math\""] }
  []
}

## Member `i` of a class, with `lit` spelled where the body uses its index.
## `member(kind, i, "${i}")` is the member `subject` generates.
pub(pkg) fn member(kind: String, i: Int, lit: String) -> String = {
  if kind == "scalar" { return "fn s${i}() -> Int = ${lit}" }
  if kind == "arith" {
    return "fn p${i}(a: Int, b: Int) -> Int = {\n  let c = a + b * 2\n  let d = c - a + ${lit}\n  let e = if d > 0 { d } else { 0 - d }\n  e + c * 3 - d\n}"
  }
  if kind == "calls" { return "fn c${i}(a: Int) -> Int = helper(a) + helper2(a, ${lit}) + helper(a + ${lit})" }
  if kind == "primitive_inferred" { return "fn ip${i}() = seed() + ${lit}" }
  if kind == "generic" {
    return "fn g${i}[T: Scale](x: T, y: T) -> Int = scale(x, ${lit}) + scale(y, 2) + scale(x, 3)"
  }
  if kind == "inferred" {
    return "fn q${i}(a: Int, b: Int) = {\n  let add = (x: Int, y: Int) => x + y\n  let twice = (f: fn(Int, Int) -> Int, x: Int) => f(x, x)\n  let t = twice(add, a + ${lit})\n  let u = twice((x, y) => x * y, b)\n  t + u\n}"
  }
  if kind == "method" {
    return "type W${i} = { v: Int }\nimpl Shine[W${i}] { fn shine(x: W${i}) -> Int = x.v * 2 + ${lit} }"
  }
  if kind == "default" {
    return "trait Buff${i}[T] { fn raw${i}(x: T) -> Int\n  fn buff${i}(x: T) -> Int = raw${i}(x) + ${lit} }"
  }
  if kind == "java" { return "fn j${i}(x: Int) -> Int !io = Math.abs(x - ${lit}) + Math.max(x, ${lit})" }
  panic("unknown subject class " ++ kind)
}

## The n members of a class, in index order.
pub(pkg) fn members(kind: String, n: Int) -> List[String] = {
  var out: List[String] = []
  for i in 0..n { out = out ++ [member(kind, i, "${i}")] }
  out
}

pub(pkg) fn render(parts: List[String]) -> String = join(parts)

## One module per class. Support declarations are a fixed O(1) prefix; the
## per-body cost divides by the number of bodies the scheduler actually ran.
pub(pkg) fn subject(kind: String, n: Int) -> String = render(support(kind) ++ members(kind, n))

# Compare actual scheduler outputs, not another implementation of checking.
pub(pkg) fn equal_bodies(actual: ModuleBodies, expected: ModuleBodies) -> Bool =
  actual.functions == expected.functions && actual.constants == expected.constants &&
  actual.impl_methods == expected.impl_methods && actual.trait_defaults == expected.trait_defaults &&
  actual.tests == expected.tests && actual.cx.diags == expected.cx.diags &&
  actual.cx.syms == expected.cx.syms && actual.cx.decl_slots == expected.cx.decl_slots &&
  actual.cx.frame == expected.cx.frame && actual.cx.fns == expected.cx.fns &&
  actual.cx.alias_resolved == expected.cx.alias_resolved &&
  actual.cx.current_tparams == expected.cx.current_tparams &&
  actual.cx.current_eff_vars == expected.cx.current_eff_vars &&
  actual.cx.current_tparam_bounds == expected.cx.current_tparam_bounds &&
  actual.cx.function_reads == expected.cx.function_reads && actual.cx.body_writes == expected.cx.body_writes &&
  actual.cx.owner == expected.cx.owner && actual.cx.owner_lo == expected.cx.owner_lo &&
  actual.cx.owner_decl == expected.cx.owner_decl && actual.cx.in_test == expected.cx.in_test &&
  actual.cx.const_cutoff == expected.cx.const_cutoff && actual.cx.loop_jumps == expected.cx.loop_jumps &&
  actual.cx.take_cell == expected.cx.take_cell

pub(pkg) fn body_count(b: ModuleBodies) -> Int =
  len(b.functions) + len(b.constants) + len(b.impl_methods) + len(b.trait_defaults) + len(b.tests)
