# Compiled only with the frozen pre-extraction scheduler in a private checker.
use front/parser
use check/checker.{
  ModuleBodies, BodyExecutor, check_module_headers, reference_module_body_products,
  cold_body_executor, execute_module_bodies
}
use check/cx.{Cx, cx_new}
use check/semantic_reads
use std/map

pub(pkg) type Comparison = { old: ModuleBodies, current: ModuleBodies }
pub(pkg) fn sample_count(xs: List[Comparison]) -> Int = len(xs)
pub(pkg) fn sample_at(xs: List[Comparison], index: Int) -> Comparison = xs[index]

pub fn main() -> Unit !io = {
  let pairs = samples()
  if sample_count(pairs) != 16 { panic("expected sixteen scheduler pairs") }
  let first = sample_at(pairs, 0)
  if len(first.old.cx.diags) != 0 { panic("clean scheduler fixture has diagnostics") }
}

# The production scheduler under an executor that only appends each role it
# was handed to its state and otherwise does what the cold executor does. The
# trace is the evidence that every role was reached through the executor seam
# (`BodyExecutor`), and the products are what the oracle compares.
fn tracing() -> BodyExecutor[List[String]] = {
  let cold: BodyExecutor[List[String]] = cold_body_executor()
  BodyExecutor {
    function: (state, cx, d, sig) => {
      let (_, next, tree) = cold.function(state, cx, d, sig)
      (state ++ ["fn " ++ d.name], next, tree)
    },
    inferred_body: (state, cx, d, sig) => {
      let (_, next, tree) = cold.inferred_body(state, cx, d, sig)
      (state ++ ["infer " ++ d.name], next, tree)
    },
    constant: (state, cx, d, ty, visible) => {
      let (_, next, tree) = cold.constant(state, cx, d, ty, visible)
      (state ++ ["const " ++ d.name], next, tree)
    },
    method: (state, cx, trait_name, subject, d, sig) => {
      let (_, next, tree) = cold.method(state, cx, trait_name, subject, d, sig)
      (state ++ ["method " ++ d.name], next, tree)
    },
    default_body: (state, cx, tid, d, sig, body) => {
      let (_, next, tree) = cold.default_body(state, cx, tid, d, sig, body)
      (state ++ ["default " ++ d.name], next, tree)
    },
    test_body: (state, cx, name, body) => {
      let (_, next, tree) = cold.test_body(state, cx, name, body)
      (state ++ ["test " ++ name], next, tree)
    }
  }
}

pub(pkg) fn samples() -> List[Comparison] !io = {
  let cases = [
    "fn caller() = leaf()\nfn leaf() = 2\nconst A: Int = 1\nconst B: Int = A\nfn add(x: Int = 7) -> Int = x + B\ntest \"result\" { assert add() == 8 }\n",
    "type Coin = { v: Int }\ntrait Waxy[T] { fn wax(x: T) -> Int\n fn buff(x: T) -> Int = wax(x) + 1 }\nimpl Waxy[Coin] { fn wax(x: Coin) -> Int = x.v }\n",
    "fn recursive() = recursive()\nfn bad() -> Int = true\nfn good() -> Int = 1\n",
    "const A: Int = B\nconst B: Int = 2\nfn bad(x: Int = false) -> Int = x\n",
    "fn duplicate() -> Int = 1\nfn duplicate() -> Int = 2\n",
    "fn closure(x: Int) -> Int = { let f = () => x + 1\n f() }\n",
    "fn first() = second()\nfn second() = first()\ntest \"failure\" { missing() }\n",
    ""
  ]
  var results: List[Comparison] = []
  var traced = 0
  for source in cases {
    let (syntax, ds) = parser.parse_module(source)
    if len(ds) != 0 { panic("scheduler fixture must parse") }
    for logging in [false, true] {
      let base = cx_new(Some("sample"), Some("sample.dawn"))
      let start = Cx { ..base, function_reads: if logging { Some([semantic_reads.StdModuleMode(true)]) } else { None } }
      let headers = check_module_headers(start, syntax, map.empty())
      let old = reference_module_body_products(headers)
      let (trace, current) = execute_module_bodies(headers, [], tracing())
      traced = traced + len(trace)
      results = results ++ [Comparison { old: old, current: current }]
    }
  }
  if traced == 0 { panic("the scheduler reached no body through the executor") }
  results
}
