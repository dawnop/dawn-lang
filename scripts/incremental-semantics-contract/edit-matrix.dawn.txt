# Count what production replay does with ten kinds of real edits.
#
# bench-replay has exactly one edit, a comment line prepended to the file, so
# its hit counts say nothing about the edits a person actually makes. This
# subject applies each edit of the matrix to the shared Scale1000 workloads
# (replay-workloads.dawn.txt) and reports the production counters of three
# revisions: before -> after through `replay`, the same step through
# `replay_and_record`, and after -> before from the renewed cache. Every step
# is compared with a cold check of the same revision, body by body.
#
# It judges nothing itself. Predictions live in edit-matrix.py, which owns the
# derivation; this side only observes, so a mutated compiler produces rows
# the driver can reject instead of a panic that hides which cell moved. No
# timers: the machine is shared and counts are the whole product of this cut.
# Driven by edit-matrix.py; stdout is strict `meta`/`step` TSV.
use std/map
use std/str
use check/checker
use check/checker.{ModuleHeaders, ModuleBodies}
use check/cx.{Cx, cx_new, ModExports}
use check/body_execution
use check/scalar_replay
use check/identity
use check/identity.{ModuleKey}
use check/source_snapshot
use check/source_snapshot.{Snapshot}
use jvm/jreflect.{jsig_real}
use contract/workloads
use contract/workloads.{equal_bodies, body_count}
use java "java.lang.Integer"

# ---------------------------------------------------------------- revisions

fn replace_at(parts: List[String], index: Int, value: String) -> List[String] = {
  var out: List[String] = []
  var i = 0
  for p in parts {
    out = out ++ [if i == index { value } else { p }]
    i = i + 1
  }
  out
}

fn insert_at(parts: List[String], index: Int, value: String) -> List[String] = {
  var out: List[String] = []
  var i = 0
  for p in parts {
    if i == index { out = out ++ [value] }
    out = out ++ [p]
    i = i + 1
  }
  out
}

fn without(parts: List[String], index: Int) -> List[String] = {
  var out: List[String] = []
  var i = 0
  for p in parts {
    if i != index { out = out ++ [p] }
    i = i + 1
  }
  out
}

fn swap_next(parts: List[String], index: Int) -> List[String] =
  replace_at(replace_at(parts, index, parts[index + 1]), index + 1, parts[index])

## One space more between two tokens inside the declaration: after its `=`.
fn spaced(decl: String) -> String = match str.split_once(decl, " = ") {
  Some((head, rest)) -> head ++ " =  " ++ rest
  None -> panic("no ` = ` to widen in " ++ decl)
}

## An inferred function with two inferred callers, appended to the class. The
## workloads have no inferred function whose return type can change without
## turning its callers into type errors (`seed() + i` needs an Int), so the
## signature edit brings its own pass-through callers.
fn probe(value: String) -> List[String] =
  ["fn edit_probe() = ${value}", "fn edit_probe_a() = edit_probe()", "fn edit_probe_b() = edit_probe()"]

## The (before, after) texts of one cell. The edited member is the middle one.
fn revisions(kind: String, edit: String, n: Int) -> (String, String) = {
  let head = workloads.support(kind)
  let body = workloads.members(kind, n)
  let m = n / 2
  let base = workloads.render(head ++ body)
  match edit {
    "identical" -> (base, base)
    "shift" -> (base, "# edited elsewhere in the file\n" ++ base)
    "body_one" -> (base, workloads.render(head ++ replace_at(body, m, workloads.member(kind, m, "${m + 1}"))))
    "body_one_type_error" -> (base, workloads.render(head ++ replace_at(body, m, workloads.member(kind, m, "false"))))
    "inferred_return" -> (workloads.render(head ++ body ++ probe("1")), workloads.render(head ++ body ++ probe("true")))
    "ws_between" -> (base, workloads.render(head ++ replace_at(body, m, "\n" ++ body[m])))
    "ws_inside" -> (base, workloads.render(head ++ replace_at(body, m, spaced(body[m]))))
    "insert_decl" -> (base, workloads.render(head ++ insert_at(body, m, workloads.member(kind, n, "${n}"))))
    "delete_decl" -> (base, workloads.render(head ++ without(body, m)))
    "reorder" -> (base, workloads.render(head ++ swap_next(body, m)))
    _ -> panic("unknown edit " ++ edit)
  }
}

# ---------------------------------------------------------------- checking

fn empty_env() -> Map[String, ModExports] = map.empty()

fn base_cx() -> Cx !io = Cx { ..cx_new(Some("sample"), Some("sample.dawn")), jsig: jsig_real() }

fn scope_key() -> ModuleKey = identity.module_key("bench", "sample", "sample", "sample.dawn")

fn snapshot_of(text: String) -> Snapshot = match source_snapshot.of(text) {
  Some(s) -> s, None -> panic("subject parse")
}

## Header diagnostics would send every body cold for a reason no edit here
## intends, so they stop the run rather than become a row.
fn headers_of(s: Snapshot) -> ModuleHeaders !io = {
  let h = checker.check_module_headers(base_cx(), source_snapshot.syntax(s), empty_env())
  if h.cx.diags != [] { panic("subject headers have diagnostics") }
  h
}

## Bodies no replay producer claims by role, whatever their shape.
fn by_role(b: ModuleBodies) -> Int =
  len(b.constants) + len(b.impl_methods) + len(b.trait_defaults) + len(b.tests)

fn flag(b: Bool) -> String = if b { "1" } else { "0" }

fn step(kind: String, edit: String, label: String, cold: ModuleBodies, r: scalar_replay.Replayed,
  visited: Int, refused: Int, admitted: Int, equal: Bool) -> Unit !io =
  println("step\t${kind}\t${edit}\t${label}\t${body_count(cold)}\t${by_role(cold)}\t${len(cold.cx.diags)}\t" ++
      "${r.checked}\t${r.reused}\t${r.cold_unadmitted}\t${r.cold_rejected}\t${visited}\t${refused}\t" ++
      "${admitted}\t${flag(equal)}")

fn run_cell(kind: String, edit: String, n: Int) -> Unit !io = {
  let (before_text, after_text) = revisions(kind, edit, n)
  let before = snapshot_of(before_text)
  let after = snapshot_of(after_text)
  let old = headers_of(before)
  let next = headers_of(after)
  let old_cold = checker.check_module_body_products(old)
  let next_cold = checker.check_module_body_products(next)
  if old_cold.cx.diags != [] { panic("before revision has body diagnostics: ${kind} ${edit}") }
  let saved = body_execution.record(scope_key(), old)
  if not equal_bodies(saved.bodies, old_cold) { panic("recording differs from cold: ${kind} ${edit}") }
  let admitted = scalar_replay.admit(scope_key(), old, before, saved)
  let first = scalar_replay.admitted_count(admitted)
  let replayed = scalar_replay.replay(admitted, next, after)
  step(kind, edit, "replay", next_cold, replayed, 0 - 1, 0 - 1, first,
    equal_bodies(replayed.bodies, next_cold))
  let renewed = scalar_replay.replay_and_record(admitted, next, after)
  let second = scalar_replay.admitted_count(renewed.admitted)
  step(kind, edit, "renew", next_cold, renewed.replayed, renewed.visited, renewed.capture_refused, second,
    equal_bodies(renewed.replayed.bodies, next_cold))
  let back = scalar_replay.replay_and_record(renewed.admitted, old, before)
  step(kind, edit, "back", old_cold, back.replayed, back.visited, back.capture_refused,
    scalar_replay.admitted_count(back.admitted), equal_bodies(back.replayed.bodies, old_cold))
}

pub fn main() -> Unit !io = {
  let a = args()
  if len(a) != 3 { panic("usage: <classes,...> <edits,...> <n>") }
  let n = Integer.parseInt(a[2])
  if n < 4 { panic("n must be at least 4") }
  println("meta\tn\t${n}")
  for kind in str.split(a[0], ",") {
    for edit in str.split(a[1], ",") { run_cell(kind, edit, n) }
  }
  println("meta\tdone\t1")
}
