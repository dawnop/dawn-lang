# Preserve the private probe interface while delegating candidate admission
# to the production declaration index. This does not establish cache validity.
use front/ast.{FnDecl, Module, DFn}
use check/identity as keys
use check/identity.{Named, FunctionDecl}

pub(pkg) type ModuleKey = { world: String, logical_module: String, emission_owner: String, source: String }
pub(pkg) type FunctionKey = { scope: ModuleKey, name: String }

pub(pkg) fn function_key(scope: ModuleKey, declarations: List[FnDecl], name: String) -> Option[FunctionKey] = {
  let owner = keys.module_key(scope.world, scope.logical_module, scope.emission_owner, scope.source)
  let m = Module { decls: map(declarations, f => DFn(f)) }
  for e in keys.declarations(owner, m) {
    if e.key.path == [Named(FunctionDecl, name)] { return Some(FunctionKey { scope: scope, name: name }) }
  }
  None
}
