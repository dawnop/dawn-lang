# Run entry validation on actual scheduler inputs, then cold-check the original
# input. Returned complete products are compared independently by the host.
use std/list
use std/str
use check/function_entry_proof as proof
use check/cx as proof_cx
use front/ast.{FnDecl, TypeParamDecl, Bound}
use check/tast.{TFun}
use check/body_product
use check/body_admit.{View}

type ProofState = { saved: Map[String, Product], accepted: Int, refused: Int, forge: Bool }
pub(pkg) type ProofSuite = { pairs: List[Pair], accepted: Int, refused: Int }

fn checked_proof(cx: Cx, d: FnDecl, sig: Sig, product: Product) -> Option[proof.Proof] !io = {
  let before = GenericTrace.count()
  let result = proof.prove(cx, d, sig, product)
  if GenericTrace.count() != before { panic("bounded entry proof executed body checker") }
  result
}

fn refused(cx: Cx, d: FnDecl, sig: Sig, product: Product, label: String) -> Unit !io =
  match checked_proof(cx, d, sig, product) {
    None -> ()
    Some(_) -> panic("bounded entry proof accepted " ++ label)
  }

fn forged_cases(cx: Cx, d: FnDecl, sig: Sig, product: Product) -> Unit !io = {
  var changed = product.symbols
  let first = list.filter(changed, row => row.key == product.tree.dict_syms[0])[0]
  let symbol = first.value.expect("proof symbol")
  changed = map(changed, row => if row.key == first.key {
    Change { ..row, value: Some(Sym { ..symbol, dhi: symbol.dhi + 1 }) }
  } else { row })
  refused(cx, d, sig, BodyProduct { ..product, symbols: changed }, "dictionary metadata")
  changed = map(product.symbols, row => if row.key == product.tree.param_syms[0] {
    let value = row.value.expect("proof parameter")
    Change { ..row, value: Some(Sym { ..value, mutable: true }) }
  } else { row })
  refused(cx, d, sig, BodyProduct { ..product, symbols: changed }, "parameter role")
  let params = product.tree.param_syms
  refused(cx, d, sig, BodyProduct { ..product,
    tree: TFun { ..product.tree, param_syms: list.reverse(params) } }, "parameter order")
  refused(cx, d, sig, BodyProduct { ..product,
    frame: Frame { ..product.frame, dict_syms: map.empty() } }, "dictionary frame")
  refused(cx, d, sig, BodyProduct { ..product, owner_decl: product.owner_decl + 1 }, "owner identity")
  refused(cx, d, sig, BodyProduct { ..product, owner_slots: product.owner_slots + 1 }, "allocation interval")
  let writes = product.body_writes.expect("proof writes")
  refused(cx, d, sig, BodyProduct { ..product, body_writes: Some(list.drop(writes, 1)) }, "entry journal")
  refused(cx, d, sig, BodyProduct { ..product, function_reads: None }, "missing reads")
  refused(cx, d, sig, BodyProduct { ..product, function_reads: Some([]) }, "entry read prefix")
  refused(cx, d, sig, BodyProduct { ..product,
    function_reads: Some([semantic_reads.StdModuleMode(false)] ++ product.function_reads.expect("proof reads")) },
    "borrowed entry read prefix")
  let binder = match sig.tparams[0] { TyVar(_, id) -> id, _ -> panic("proof binder") }
  refused(Cx { ..cx, current_tparam_bounds: map.insert(cx.current_tparam_bounds, binder, []) },
    d, sig, product, "header bounds")
  let tp = d.tparams[0]
  let bound = tp.bounds[0]
  let altered = TypeParamDecl { ..tp, bounds: [Bound { ..bound, name: "MissingBound" }] }
  refused(cx, FnDecl { ..d, tparams: [altered] ++ list.drop(d.tparams, 1) }, sig, product, "written bounds")
  let param = d.params[0]
  refused(Cx { ..cx, module_aliases: map.insert(cx.module_aliases, param.name, "provider") },
    d, sig, product, "entry diagnostics")
}

fn proof_step(state: ProofState, cx: Cx, d: FnDecl, sig: Sig) -> (ProofState, Cx, TFun) !io = {
  let product = map.get(state.saved, d.name).expect("proof saved function")
  let before_slots = proof_cx.slot_of(cx)
  let before_reads = cx.function_reads
  let before_writes = cx.body_writes
  let candidate = checked_proof(cx, d, sig, product)
  var accepted = state.accepted
  var rejected = state.refused
  match candidate {
    Some(value) -> {
      accepted = accepted + 1
      let entry = proof.context(value)
      if map.get(entry.current_tparam_bounds, -12345) != map.get(cx.current_tparam_bounds, -12345) {
        panic("bounded entry proof discarded unrelated header bounds")
      }
      let view = View { symbols: entry.syms, callee: (owner, name, builtin, trait_id) => {
        for (_, found) in map.entries(entry.fns) {
          if found.owner == owner && found.name == name && found.is_builtin == builtin && found.trait_id == trait_id {
            return Some(found)
          }
        }
        None
      } }
      let projected = body_product.project(view, product).expect("bounded entry source projection")
      match checked_proof(cx, d, sig, projected) {
        None -> panic("bounded entry proof rejected relative projection")
        Some(_) -> ()
      }
      if proof.parameters(value) != product.tree.param_syms || proof.dictionaries(value) != product.tree.dict_syms {
        panic("bounded entry proof lost ordered allocations")
      }
      if entry.function_reads == cx.function_reads || entry.body_writes == cx.body_writes {
        panic("bounded entry proof borrowed observation prefix")
      }
      for fact in product.function_reads.expect("proof recorded reads") {
        match fact {
          semantic_reads.DictionarySymbol(binder, trait_id, expected) -> {
            if map.get(entry.frame.dict_syms, (binder, trait_id)) != expected {
              panic("bounded entry proof lost caller binder substitution")
            }
          }
          _ -> ()
        }
      }
      if state.forge { forged_cases(cx, d, sig, product) }
    }
    None -> { rejected = rejected + 1 }
  }
  if proof_cx.slot_of(cx) != before_slots || cx.function_reads != before_reads || cx.body_writes != before_writes {
    panic("bounded entry proof advanced live scheduler")
  }
  let (after, tree) = checker.check_fn(cx, d, sig)
  (ProofState { ..state, accepted: accepted, refused: rejected }, after, tree)
}

pub(pkg) fn proof_cases() -> ProofSuite !io = {
  var pairs: List[Pair] = []
  var accepted = 0
  var rejected = 0
  for sample in samples() {
    var saved: Map[String, Product] = map.empty()
    for product in sample.products { saved = map.insert(saved, product.tree.name, product) }
    for moved in [false, true] {
      let source = if moved { "# relative source move\n\n" ++ sample.source } else { sample.source }
      let snapshot = source_snapshot.of(source).expect("proof source")
      let base = cx_new(Some("sample"), Some("sample.dawn"))
      let initial = checker.check_module_headers(base, source_snapshot.syntax(snapshot), map.empty())
      for observing in [false, true] {
        let headers = ModuleHeaders { ..initial, cx: Cx { ..initial.cx,
          current_tparam_bounds: map.insert(initial.cx.current_tparam_bounds, -12345, []),
          function_reads: if observing { Some([semantic_reads.StdModuleMode(false)]) } else { None },
          body_writes: if observing { Some([SignatureKey("outside")]) } else { None } } }
        let cold: BodyExecutor[ProofState] = checker.cold_body_executor()
        let executor = BodyExecutor { ..cold, function: (state, cx, d, sig) => proof_step(state, cx, d, sig) }
        let start = ProofState { saved: saved, accepted: 0, refused: 0, forge: true }
        let (state, actual) = checker.execute_module_bodies(headers, start, executor)
        if state.accepted != len(sample.products) || state.refused != 0 {
          panic("bounded entry proof rejected canonical " ++ sample.label)
        }
        accepted = accepted + state.accepted
        pairs = pairs ++ [Pair { cold: checker.check_module_body_products(headers), actual: actual }]
      }
    }
    if sample.label == "multiple-bounds" {
      let source = str.replace(sample.source, "Scale + Stamp", "Stamp + Scale")
      let snapshot = source_snapshot.of(source).expect("proof reordered bounds")
      let headers = checker.check_module_headers(cx_new(Some("sample"), Some("sample.dawn")),
        source_snapshot.syntax(snapshot), map.empty())
      let cold: BodyExecutor[ProofState] = checker.cold_body_executor()
      let executor = BodyExecutor { ..cold, function: (state, cx, d, sig) => proof_step(state, cx, d, sig) }
      let (state, actual) = checker.execute_module_bodies(headers,
        ProofState { saved: saved, accepted: 0, refused: 0, forge: false }, executor)
      if state.accepted != 0 || state.refused != 1 { panic("bounded entry proof accepted reordered header bounds") }
      rejected = rejected + state.refused
      pairs = pairs ++ [Pair { cold: checker.check_module_body_products(headers), actual: actual }]
    }
  }
  let support = "trait Scale[T] { fn scale(x: T, k: Int) -> Int }\n"
  for source in [
    support ++ "fn excluded[T: Scale](x: T, y: T) -> Int !io = scale(x, 1)\n",
    support ++ "fn excluded[T: Scale](x: T, y: T, k: Int = 1) -> Int = scale(x, k)\n",
    support ++ "fn excluded[T: Scale](x: T, y: T) -> Int = {\n let k = 1\n scale(x, k)\n}\n"
  ] {
    let scope = identity.module_key("generic-trace", "sample", "sample", "sample.dawn")
    let snapshot = source_snapshot.of(source).expect("proof exclusion source")
    let headers = checker.check_module_headers(cx_new(Some("sample"), Some("sample.dawn")),
      source_snapshot.syntax(snapshot), map.empty())
    let recorded = body_execution.record(scope, headers)
    if recorded.bodies.cx.diags != [] || recorded.refused != 0 { panic("bounded entry excluded fixture failed capture") }
    var saved: Map[String, Product] = map.empty()
    for item in recorded.entries {
      match item { SavedFunction(_, FunctionInput(d, _, _), product) -> { saved = map.insert(saved, d.name, product) }, _ -> () }
    }
    let cold: BodyExecutor[ProofState] = checker.cold_body_executor()
    let executor = BodyExecutor { ..cold, function: (state, cx, d, sig) => proof_step(state, cx, d, sig) }
    let (state, actual) = checker.execute_module_bodies(headers,
      ProofState { saved: saved, accepted: 0, refused: 0, forge: false }, executor)
    if state.accepted != 0 || state.refused != 1 { panic("bounded entry proof widened excluded class") }
    rejected = rejected + state.refused
    pairs = pairs ++ [Pair { cold: checker.check_module_body_products(headers), actual: actual }]
  }
  ProofSuite { pairs: pairs, accepted: accepted, refused: rejected }
}

pub(pkg) fn proof_pair_count(suite: ProofSuite) -> Int = len(suite.pairs)
pub(pkg) fn proof_pair_at(suite: ProofSuite, index: Int) -> Pair = suite.pairs[index]
pub(pkg) fn proof_accept_count(suite: ProofSuite) -> Int = suite.accepted
pub(pkg) fn proof_refuse_count(suite: ProofSuite) -> Int = suite.refused
