# Malformed products are tested through actual admission/replay, preserving the
# complete original read log unless the read itself is the mutated evidence.
use std/list
use std/io
use std/str
use check/tast.{TExpr, TFun, XBinary, XCallFn, XLocal, WForward}
use check/semantic_reads.{UnificationRead, InferenceBindings}
use driver/analyze.{Program, CheckedMod, prepare_standalone, prepared_result, analyze_program}
use driver/incremental
use driver/stdlib.{load_std, std_load_error_text}
use check/jsig.{jsig_refused, refused_probe}
use ir/interp.{ct_default}

pub(pkg) type ProgramPair = { cold: Program, actual: Program }
pub(pkg) type BoundedSuite = { bodies: List[Pair], programs: List[ProgramPair] }

fn first_argument(e: TExpr) -> TExpr = match e {
  XBinary(_, left, _, _, _, _, _, _, _) -> first_argument(left)
  XCallFn(_, _, args, _, _, _, _, _, _) -> args[0]
  _ -> panic("bounded replay fixture needs a call")
}

fn changed_call(e: TExpr, p: Product, mode: String) -> TExpr = match e {
  XBinary(op, left, right, witness, olo, ohi, lo, hi, ty) ->
    XBinary(op, changed_call(left, p, mode), right, witness, olo, ohi, lo, hi, ty)
  XCallFn(owner, name, args, witnesses, evidence, trait_id, lo, hi, ty) -> {
    if mode == "witness role" {
      return XCallFn(owner, name, args, [WForward(p.tree.dict_syms[1])], evidence, trait_id, lo, hi, ty)
    }
    if mode == "missing witness" {
      return XCallFn(owner, name, args, [], evidence, trait_id, lo, hi, ty)
    }
    if mode == "callee owner" {
      return XCallFn(Some("forged-owner"), name, args, witnesses, evidence, trait_id, lo, hi, ty)
    }
    if mode == "argument unification" {
      return XCallFn(owner, name, [args[0], args[0]], witnesses, evidence, trait_id, lo, hi, ty)
    }
    if mode == "argument role" {
      let other = list.filter(p.symbols, row => row.key == p.tree.param_syms[1])[0].value.expect("other parameter")
      let argument = match args[0] {
        XLocal(id, local_name, alo, ahi, _) -> XLocal(id, local_name, alo, ahi, other.ty)
        _ -> panic("bounded replay fixture needs a local argument")
      }
      return XCallFn(owner, name, [argument] ++ list.drop(args, 1), [WForward(p.tree.dict_syms[1])], evidence, trait_id, lo, hi, ty)
    }
    e
  }
  _ -> panic("bounded replay fixture malformed call graph")
}

fn changed_product(p: Product, mode: String) -> Product = {
  if mode == "root result" { return BodyProduct { ..p, tree: TFun { ..p.tree, body: first_argument(p.tree.body) } } }
  if mode == "entry frame" { return BodyProduct { ..p, frame: Frame { ..p.frame, dict_syms: map.empty() } } }
  if mode == "unification fact" {
    let reads = map(p.function_reads.expect("bounded replay reads"), read => match read {
      semantic_reads.UnifiedTypes(q) -> semantic_reads.UnifiedTypes(UnificationRead { ..q,
        after: InferenceBindings { ..q.after, types: [] } })
      _ -> read
    })
    return BodyProduct { ..p, function_reads: Some(reads) }
  }
  if mode == "unknown fact" {
    return BodyProduct { ..p, function_reads: Some(p.function_reads.expect("bounded replay reads") ++ [semantic_reads.StdModuleMode(false)]) }
  }
  BodyProduct { ..p, tree: TFun { ..p.tree, body: changed_call(p.tree.body, p, mode) } }
}

pub(pkg) fn bounded_cases() -> BoundedSuite !io = {
  var pairs: List[Pair] = []
  let scope = identity.module_key("generic-trace", "sample", "sample", "sample.dawn")
  let sample = samples()[2]
  let snapshot = source_snapshot.of(sample.source).expect("bounded replay source")
  let initial = checker.check_module_headers(cx_new(Some("sample"), Some("sample.dawn")),
    source_snapshot.syntax(snapshot), map.empty())
  let recorded = body_execution.record(scope, initial)
  for mode in ["root result", "witness role", "missing witness", "callee owner", "argument unification",
    "argument role", "entry frame", "unification fact", "unknown fact"] {
    let entries = map(recorded.entries, item => match item {
      SavedFunction(key, FunctionInput(d, sig, inferred), p) ->
        SavedFunction(key, FunctionInput(d, sig, inferred), changed_product(p, mode))
      _ -> item
    })
    let admitted = scalar_replay.admit(scope, initial, snapshot, Recorded { ..recorded, entries: entries })
    GenericTrace.reset()
    let actual = scalar_replay.replay(admitted, initial, snapshot)
    if actual.reused != 0 || actual.checked != 2 || GenericTrace.count() != 2 {
      panic("bounded replay rejected " ++ mode)
    }
    pairs = pairs ++ [Pair { cold: checker.check_module_body_products(initial), actual: actual.bodies }]
  }
  # Enclosing logs must survive actual renewal, not just detached entry tests.
  for observing in [false, true] {
    let headers = ModuleHeaders { ..initial, cx: Cx { ..initial.cx,
      function_reads: if observing { Some([semantic_reads.StdModuleMode(false)]) } else { None },
      body_writes: if observing { Some([SignatureKey("outside")]) } else { None } } }
    var admitted = scalar_replay.admit(scope, headers, snapshot, body_execution.record(scope, headers))
    for prefix in ["# first move\n", "# second move\n\n", "# third move\n\n\n"] {
      let moved = source_snapshot.of(prefix ++ sample.source).expect("bounded replay moved")
      let current = checker.check_module_headers(cx_new(Some("sample"), Some("sample.dawn")), source_snapshot.syntax(moved), map.empty())
      let candidate = ModuleHeaders { ..current, cx: Cx { ..current.cx,
        function_reads: headers.cx.function_reads, body_writes: headers.cx.body_writes } }
      GenericTrace.reset()
      let renewed = scalar_replay.replay_and_record(admitted, candidate, moved)
      if renewed.replayed.reused != 1 || renewed.replayed.checked != 1 || GenericTrace.count() != 1 ||
        renewed.capture_refused != 0 || scalar_replay.admitted_count(renewed.admitted) != 1 {
        panic("bounded replay prefixed renewal counts")
      }
      pairs = pairs ++ [Pair { cold: checker.check_module_body_products(candidate), actual: renewed.replayed.bodies }]
      admitted = renewed.admitted
    }
  }
  BoundedSuite { bodies: pairs, programs: prepared_bounded_cases() }
}

fn prepared_bounded_cases() -> List[ProgramPair] !io = io.with_fs_real(() => io.with_env_real(() => {
  let std = match load_std("std") { Ok(value) -> value, Err(error) -> panic(std_load_error_text(error)) }
  let support = "trait Scale[T] { fn scale(x: T, k: Int) -> Int }\ntype Coin = { value: Int }\n"
  let implementation = "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Int = x.value * k }\n"
  let changed = "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Int = x.value * k + 1 }\n"
  let first = "fn g0[T: Scale](x: T, y: T) -> Int = scale(x, 0) + scale(y, 2) + scale(x, 3)\n"
  let second = "fn g1[T: Scale](x: T, y: T) -> Int = scale(x, 1) + scale(y, 2) + scale(x, 3)\n"
  let oracle = jsig_refused()
  var session = incremental.new_with_body_cache(std, ct_default(), oracle, refused_probe, 8, 100000, 20)
  var pairs: List[ProgramPair] = []
  var index = 0
  for text in [support ++ implementation ++ first ++ second,
    "# moved\n" ++ support ++ implementation ++ first ++ second,
    support ++ changed ++ second ++ first,
    "# renewed\n" ++ support ++ changed ++ first ++ second] {
    let input = prepare_standalone("<lsp>", text)
    GenericTrace.reset()
    let actual = incremental.analyze_prepared(session, input)
    if actual.stats.checked_bodies != (if index == 0 { 3 } else { 1 }) ||
      actual.stats.reused_bodies != (if index == 0 { 0 } else { 2 }) ||
      GenericTrace.count() != actual.stats.checked_bodies || len(actual.program.diags) != 0 {
      panic("bounded replay prepared Session counts")
    }
    let cold = analyze_program(prepared_result(input), std, ct_default(), oracle)
    pairs = pairs ++ [ProgramPair { cold: cold, actual: actual.program }]
    session = actual.session
    index = index + 1
  }
  # Positional calls remain well typed when only the trait parameter spelling
  # changes, but their canonical signature facts must still be replaced.
  let two_traits = support ++ "trait Stamp[T] { fn stamp(x: T) -> Int }\n"
  let renamed = str.replace(two_traits, "scale(x: T, k: Int)", "scale(x: T, factor: Int)")
  let stamp = "impl Stamp[Coin] { fn stamp(x: Coin) -> Int = x.value }\n"
  let more_bounds = str.replace(first, "T: Scale", "T: Scale + Stamp")
  let bad_header = str.replace(renamed, "factor: Int) -> Int", "factor: Int) -> Bool")
  let bad_impl = "impl Scale[Coin] { fn scale(x: Coin, k: Int) -> Bool = true }\n"
  session = incremental.new_with_body_cache(std, ct_default(), oracle, refused_probe, 8, 100000, 20)
  for (text, checked, reused, broken) in [
    (two_traits ++ implementation ++ stamp ++ first ++ second, 4, 0, false),
    (renamed ++ implementation ++ stamp ++ first ++ second, 4, 0, false),
    ("# renamed recovery\n" ++ renamed ++ implementation ++ stamp ++ first ++ second, 2, 2, false),
    (renamed ++ implementation ++ stamp ++ more_bounds ++ second, 3, 1, false),
    ("# bounds recovery\n" ++ renamed ++ implementation ++ stamp ++ more_bounds ++ second, 2, 2, false),
    (bad_header ++ bad_impl ++ stamp ++ more_bounds ++ second, 4, 0, true),
    (renamed ++ implementation ++ stamp ++ more_bounds ++ second, 4, 0, false),
    ("# repaired recovery\n" ++ renamed ++ implementation ++ stamp ++ more_bounds ++ second, 2, 2, false)
  ] {
    let input = prepare_standalone("<lsp>", text)
    GenericTrace.reset()
    let actual = incremental.analyze_prepared(session, input)
    if actual.stats.checked_bodies != checked || actual.stats.reused_bodies != reused ||
      GenericTrace.count() != checked || (len(actual.program.diags) != 0) != broken {
      panic("bounded replay real dependency transition")
    }
    let cold = analyze_program(prepared_result(input), std, ct_default(), oracle)
    pairs = pairs ++ [ProgramPair { cold: cold, actual: actual.program }]
    session = actual.session
  }
  pairs
}))

pub(pkg) fn bounded_body_count(suite: BoundedSuite) -> Int = len(suite.bodies)
pub(pkg) fn bounded_body_at(suite: BoundedSuite, index: Int) -> Pair = suite.bodies[index]
pub(pkg) fn bounded_program_count(suite: BoundedSuite) -> Int = len(suite.programs)
pub(pkg) fn bounded_program_at(suite: BoundedSuite, index: Int) -> ProgramPair = suite.programs[index]
pub(pkg) fn bounded_module_count(p: Program) -> Int = len(p.modules)
pub(pkg) fn bounded_module_at(p: Program, index: Int) -> CheckedMod = p.modules[index]
