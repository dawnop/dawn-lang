# Only compiled after the frozen loop is injected into a private driver.
# Written to the compiler copy as contract/reference; SemanticSnapshot drives
# `samples` by reflection, and `main` exists so the fixture entry reaches it.
use std/io
use front/parser
use front/lexer
use driver/analyze.{Program, LoadedModule, LoadResult, LocDiag, analyze_program, analyze_reference}
use driver/stdlib.{load_std, std_load_error_text}
use check/jsig.{jsig_refused}
use ir/interp.{ct_default, ct_fuel}
pub(pkg) type Comparison = { old: Program, current: Program }

fn input(name: String, text: String) -> LoadedModule = {
  let (m, pd, marks) = parser.parse_module_marked(text)
  LoadedModule {
    mod_path = name, class_name = name, path = name ++ ".dawn", site_path = name ++ ".dawn",
    text = text, line_starts = lexer.line_starts_of(text), m = m, pdiags = pd, marks = marks, pkg = None, entry = false
  }
}

pub(pkg) fn sample_count(xs: List[Comparison]) -> Int = len(xs)

pub(pkg) fn sample_at(xs: List[Comparison], index: Int) -> Comparison = xs[index]

pub(pkg) fn samples() -> List[Comparison] !io = {
  io.with_fs_real(() => io.with_env_real(() => {
    let std = match load_std("std") {
      Ok(s) -> s
      Err(e) -> panic(std_load_error_text(e))
    }
    let lib = input("lib",
      "pub trait Sizey[T] { fn sizey(x: T) -> Int }\n" ++
        "pub type Box = { n: Int } derive Show\n" ++
        "impl Sizey[Box] { fn sizey(b: Box) -> Int = b.n + 2 }\n" ++
        "pub fn describe[T: Sizey](x: T) -> Int = sizey(x)\n" ++
        "pub const N: Int = 40 + 2\n")
    let middle = input("middle", "pub type Other = { n: Int }\npub fn make() -> Other = Other { n = 7 }\n")
    let consumer = input("consumer",
      "use lib\nuse lib.{Box}\npub fn value() -> Int = lib.describe(Box { n = 3 })\n")
    let bad = input("bad", "pub fn bad() -> Int = false\n")
    let broken = input("broken", "pub fn (\n")
    # a statement the parser cannot read, in a module that grafts a type, has
    # a trait, an impl and a const, and a signature naming a type nobody
    # declares: only the headers of what was recovered are published, and
    # none of their diagnostics
    let torn = input("lib",
      "use middle.{Other}\n" ++
        "pub trait Sizey[T] { fn sizey(x: T) -> Int }\n" ++
        "pub type Box = { n: Int } derive Show\n" ++
        "impl Sizey[Box] { fn sizey(b: Box) -> Int = b.n + 2 }\n" ++
        "pub fn describe[T: Sizey](x: T) -> Int = sizey(x)\n" ++
        "pub fn other(o: Other) -> Int = o.n\n" ++
        "fn torn() -> Int = {\n  let = 1\n  1\n}\n" ++
        "pub fn later() -> Missing = 1\n" ++
        "pub const N: Int = 40 + 2\n")
    # a module that fails to parse without losing a declaration is checked
    # past the damaged ones: `torn` is damaged and infers a String that
    # `uses` wants as an Int, a statement the parser skipped is the only use
    # of the import, a signature names a type nobody declares in a damaged
    # declaration, and `later` and `also` are whole and wrong
    let recovered = input("recov",
      "use middle.{Other}\n" ++
        "fn torn() = {\n  let = Other { n = 1 }\n  \"text\"\n}\n" ++
        "fn uses() -> Int = torn()\n" ++
        "fn damaged_header(x: Gone) -> Int = {\n  let = 2\n  3\n}\n" ++
        "fn later() -> Int = \"not an int\"\n" ++
        "fn also() -> Int = missing_name\n" ++
        "pub fn fine() -> Int = 1\n")
    let recovered_user = input("recov_user", "use recov.{fine}\npub fn total() -> Int = fine() + 1\n")
    # an unclosed block, and a declaration the parser dropped
    let unclosed = input("unclosed",
      "fn torn() = {\n  for x in [1] {\n    x\n}\n\nfn later() -> Int = \"not an int\"\n")
    let dropped = input("dropped",
      "use middle.{Other}\n" ++
        "pub type Box = { n: Int }\nimpl Show[Box] { fn show(b: Box) -> String = \"box\" }\n" ++
        "pub fn other(o: Other) -> Int = o.n\n" ++
        "pub fn gone() -> Missing = 1\n" ++
        "fn fine() -> Int = 1\nfn broken( = 0\nfn later() -> Int = \"not an int\"\n")
    let loader = LocDiag { path = "loader", d = broken.pdiags[0] }
    var results: List[Comparison] = []
    let cases: List[LoadResult] = [
      LoadResult { modules = [lib, middle, consumer], diags = [] },
      LoadResult { modules = [bad, lib, consumer], diags = [loader] },
      LoadResult { modules = [broken, lib, bad, consumer], diags = [] },
      LoadResult { modules = [middle, lib, consumer], diags = [] },
      LoadResult { modules = [middle, torn, consumer], diags = [] },
      LoadResult { modules = [lib], diags = [] },
      LoadResult { modules = [], diags = [loader] },
      LoadResult { modules = [middle, recovered, recovered_user], diags = [] },
      LoadResult { modules = [middle, unclosed, dropped], diags = [] }
    ]
    for loaded in cases {
      for opts in [ct_default(), ct_fuel(0)] {
        let old = analyze_reference(loaded, std, opts, jsig_refused())
        let current = analyze_program(loaded, std, opts, jsig_refused())
        results = results ++ [Comparison { old = old, current = current }]
      }
    }
    let clean = analyze_reference(cases[0], std, ct_default(), jsig_refused())
    if len(clean.diags) != 0 { panic("reference clean fixture has diagnostics") }
    # std rechecking intentionally uses impls_before, not the final bundle.
    let text = match io.read_file("std/list.dawn") {
      Ok(t) -> t
      Err(e) -> panic(e.message)
    }
    let list = LoadedModule { ..input("list", text), path = "std/list.dawn" }
    let loaded = LoadResult { modules = [list], diags = [] }
    let old = analyze_reference(loaded, std, ct_default(), jsig_refused())
    let current = analyze_program(loaded, std, ct_default(), jsig_refused())
    if len(old.diags) != 0 { panic("reference std fixture has diagnostics") }
    results = results ++ [Comparison { old = old, current = current }]
    results
  }))
}

pub fn main() -> Unit !io = ()
