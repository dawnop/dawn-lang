# Setup and measured intervals are explicit. The project/standalone samples
# include all preparation; only the separate consumer excludes std/loader setup.
use std/io
use std/map
use driver/analyze
use driver/analyze.{ProjectPlan, ModuleIndex}
use driver/incremental
use driver/stdlib.{load_std, std_load_error_text}
use check/jsig.{jsig_refused, refused_probe}
use ir/interp.{ct_default}
use compiler_plan/source.{SourcePlan, SourceFile}
use java "contract.SourceParseCounts"

fn plan() -> ProjectPlan = ProjectPlan {
  source: SourcePlan { target: SourceFile("/prepared/main.dawn"), project: "/prepared",
    source_root: "/prepared", pkgs: map.empty(), java_coords: [], diags: [] },
  modules: ModuleIndex { project_files: [], by_use: map.empty() }
}

fn overlay() -> Map[String, String] = map.from([
  ("/prepared/main.dawn", "use middle\nfn value() = middle.value()\n"),
  ("/prepared/middle.dawn", "use leaf\npub fn value() -> Int = leaf.value()\n"),
  ("/prepared/leaf.dawn", "pub fn value() -> Int = 42\n")
])

fn clean(loaded: analyze.LoadResult) -> Bool = {
  if loaded.diags != [] || len(loaded.modules) != 3 { return false }
  for module in loaded.modules { if module.pdiags != [] { return false } }
  true
}

pub(pkg) fn project_on() -> Bool !io = io.with_fs_real(() => io.with_env_real(() => {
  let loaded = analyze.prepared_result(analyze.load_entries_over_prepared(plan(), ["/prepared/main.dawn"], overlay()))
  clean(loaded) && loaded.modules[0].mod_path == "leaf"
}))

pub(pkg) fn project_off() -> Bool !io = io.with_fs_real(() => io.with_env_real(() => {
  let loaded = analyze.load_entries_over(plan(), ["/prepared/main.dawn"], overlay())
  clean(loaded) && loaded.modules[2].mod_path == "main"
}))

pub(pkg) fn standalone() -> Bool = {
  let input = analyze.prepare_standalone("<lsp>", "fn leaf() = 42\n")
  let loaded = analyze.prepared_result(input)
  loaded.diags == [] && len(loaded.modules) == 1 && len(loaded.modules[0].m.decls) == 1
}

pub(pkg) fn session_consumer() -> Bool !io = io.with_fs_real(() => io.with_env_real(() => {
  let std = match load_std("std") {
    Ok(value) -> value
    Err(error) -> panic(std_load_error_text(error))
  }
  let prepared = analyze.prepare_standalone("<lsp>", "fn leaf() = 42\nfn caller() = leaf()\n")
  let owner = incremental.new_with_body_cache(std, ct_default(), jsig_refused(), refused_probe, 4, 10000, 10)
  # Only this sample excludes preparation. Neither loader sample resets here.
  SourceParseCounts.begin()
  let first = incremental.analyze_prepared(owner, prepared)
  let next = incremental.analyze_prepared(incremental.evict(first.session), prepared)
  first.program.diags == [] && first.stats.checked_bodies == 2 &&
    next.stats.checked_bodies == 2 && next.program.diags == []
}))

pub fn main() -> Unit = ()
